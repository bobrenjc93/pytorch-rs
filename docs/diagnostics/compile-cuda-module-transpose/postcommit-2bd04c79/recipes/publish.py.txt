sha = lambda raw: hashlib.sha256(raw).hexdigest()
def copy_bytes(path, name, compressed=False):
    raw=path.read_bytes(); out=dest/name; out.parent.mkdir(parents=True,exist_ok=True)
    out.write_bytes(gzip.compress(raw,mtime=0) if compressed else raw)
    if compressed: assert gzip.decompress(out.read_bytes()) == raw
    return {'sha256':sha(raw),'bytes':len(raw),'file':name}

assets = {}

    data=json.loads((receipts/(label+'.json')).read_text())
    commands[label]=data
    assets[label]=copy_bytes(receipts/(label+'.log'),'logs/'+label+'.log.gz',True)
    assert assets[label]['sha256'] == data['log_sha256']
    for key in ('production_manifest','input_manifest'):
        name=data[key]+'.json'
        copy_bytes(receipts/name,'manifests/'+name+'.gz',True)
(dest/'commands.json').write_text(json.dumps(commands,indent=2)+'\n')
copy_bytes(receipts/'provenance-audit.log','verification.json')
copy_bytes(receipts/'environment.log','environment.json')
copy_bytes(build_dir/'build-record.json','build-record.json.gz',True)
for label, name in (('build-dependencies','dependency_setup.log'),('release-build','build.log'),('origin-patch','origin.patch')):
    assets[label]=copy_bytes(build_dir/name,'logs/'+label+'.gz',True)
assets['venv-creation']=copy_bytes(work/'venv.log','logs/venv-creation.log.gz',True)

    copy_bytes(work/name,'recipes/'+name+'.txt')
(dest/'assets.json').write_text(json.dumps(assets,indent=2)+'\n')
