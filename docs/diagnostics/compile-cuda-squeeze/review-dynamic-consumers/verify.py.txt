import ctypes
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys

root = Path.cwd().resolve()
sys.path.insert(0, str(root / 'scripts'))
from evaluate_cuda_math import source_provenance

def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

build_path = root / 'target/squeeze-dynamic-review/release/build-record.json'
build = json.loads(build_path.read_text())
assert source_provenance()['source_sha256'] == build['source_sha256']
assert source_provenance()['commit'] == build['commit']
assert sys.version_info[:3] == (3, 12, 14)
assert Path(sys.prefix).resolve() == root / '.venv'
assert Path(sys.executable).resolve().is_relative_to(root / 'target/uv-python')
import torch_rs as native
assert Path(native.__file__).resolve().is_relative_to(root / '.venv')
assert sha(native._C.__file__) == build['extension_sha256']
assert sha(build['wheel_path']) == build['wheel_sha256']
for source in (root / 'python/torch_rs').rglob('*.py'):
    installed = Path(native.__file__).parent / source.relative_to(root / 'python/torch_rs')
    assert installed.read_bytes() == source.read_bytes(), source
manifest = root / 'docs/diagnostics/compile-cuda-squeeze/inputs.json'
inputs = json.loads(manifest.read_text())
changed = 'tests/test_compile_cuda_squeeze.py'
for path, digest in (inputs['files'] | inputs['compiler_sweep_modules']).items():
    if path != changed:
        assert sha(root / path) == digest, path
assert set(inputs['compiler_sweep_modules']) == {str(p.relative_to(root)) for p in root.glob('tests/test_compile*.py')}
assert native.cuda.is_available()
def runtime_paths():
    return sorted({line.split()[-1] for line in Path('/proc/self/maps').read_text().splitlines() if '/libcudart.so' in line})
paths = runtime_paths()
assert paths == [os.environ['TORCH_RS_CUDART']]
runtime = ctypes.CDLL(paths[0])
version = ctypes.c_int()
assert runtime.cudaRuntimeGetVersion(ctypes.byref(version)) == 0
import torch
assert torch.__version__ == '2.13.0+cu130'
assert torch.cuda.device_count() == 1 and torch.cuda.get_device_name(0) == 'NVIDIA H100'
assert version.value == 13000 and runtime_paths() == paths
record = {**source_provenance(), 'measurement_kind': 'precommit-review-revision',
          'status': subprocess.check_output(['git', 'status', '--porcelain'], text=True),
          'package_path': native.__file__, 'native_path': native._C.__file__,
          'native_sha256': sha(native._C.__file__), 'installed_python_matches_source': True,
          'build_record_sha256': sha(build_path), 'python': sys.version, 'executable': sys.executable,
          'input_manifest': '../inputs.json', 'input_manifest_sha256': sha(manifest),
          'changed_inputs': {changed: sha(root / changed)}, 'all_other_input_hashes_verified': True,
          'compiler_modules': len(inputs['compiler_sweep_modules']),
          'runtime': {'path': paths[0], 'sha256': sha(paths[0]), 'version': version.value,
                      'reference': torch.__version__, 'reference_cuda': torch.version.cuda}}
out = root / ('target/squeeze-dynamic-review/verification-' + sys.argv[1] + '.json')
assert not out.exists()
out.write_text(json.dumps(record, indent=2) + '\n')
print(json.dumps(record, indent=2))
