"""Verify clean commit, installed wheel, reused inputs and loaded CUDA runtime."""
import ctypes, hashlib, json, os
from pathlib import Path
import subprocess, sys
ROOT = Path.cwd().resolve()
OUT = ROOT / 'target/postcommit-e835f7f/capture'
sys.path.insert(0, str(ROOT / 'scripts'))
from evaluate_cuda_math import source_provenance

def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

assert subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip() == 'e835f7f173af4ffc7783560d9b03709660fa3141'
assert subprocess.check_output(['git','status','--porcelain'],text=True) == ''
assert Path(sys.prefix).resolve() == ROOT / '.venv'
assert Path(sys.executable).resolve().is_relative_to(ROOT / 'target/uv-python')
assert sys.version_info[:3] == (3,12,14)
build = json.loads((ROOT / 'target/postcommit-e835f7f/release/build-record.json').read_text())
assert build['clean_checkout'] and build['git_status_before_build'] == ''
assert source_provenance()['source_sha256'] == build['source_sha256']
assert build['commit'] == source_provenance()['commit']
import torch_rs as native
assert Path(native.__file__).resolve().is_relative_to(ROOT / '.venv')
assert Path(native._C.__file__).resolve().is_relative_to(ROOT / '.venv')
assert sha(native._C.__file__) == build['extension_sha256']
assert sha(build['wheel_path']) == build['wheel_sha256']
for source in (ROOT / 'python/torch_rs').rglob('*.py'):
    installed = Path(native.__file__).parent / source.relative_to(ROOT / 'python/torch_rs')
    assert source.read_bytes() == installed.read_bytes(), source
inputs_path = ROOT / 'docs/diagnostics/compile-cuda-squeeze/inputs.json'
inputs = json.loads(inputs_path.read_text())
for path, digest in (inputs['files'] | inputs['compiler_sweep_modules']).items():
    assert sha(ROOT / path) == digest, path
assert set(inputs['compiler_sweep_modules']) == {str(p.relative_to(ROOT)) for p in ROOT.glob('tests/test_compile*.py')}
assert native.cuda.is_available()
paths_before = sorted({line.split()[-1] for line in Path('/proc/self/maps').read_text().splitlines() if '/libcudart.so' in line})
assert paths_before == [os.environ['TORCH_RS_CUDART']], paths_before
runtime = ctypes.CDLL(paths_before[0])
version = ctypes.c_int()
assert runtime.cudaRuntimeGetVersion(ctypes.byref(version)) == 0
import torch
assert torch.__version__.split('+')[0] == '2.13.0'
assert torch.cuda.device_count() == 1 and torch.cuda.get_device_name(0) == 'NVIDIA H100'
assert torch.version.cuda == '13.0' and version.value == 13000
paths_after = sorted({line.split()[-1] for line in Path('/proc/self/maps').read_text().splitlines() if '/libcudart.so' in line})
assert paths_after == paths_before
report = {**source_provenance(), 'git_status': '', 'python': sys.version, 'executable': sys.executable,
          'interpreter_realpath': str(Path(sys.executable).resolve()),
          'interpreter_sha256': sha(Path(sys.executable).resolve()),
          'package_path': native.__file__, 'extension_path': native._C.__file__,
          'extension_sha256': sha(native._C.__file__), 'all_installed_python_files_match_source': True,
          'build_record': 'build-record.json', 'build_record_sha256': sha(ROOT / 'target/postcommit-e835f7f/release/build-record.json'),
          'input_manifest': '../inputs.json', 'input_manifest_sha256': sha(inputs_path),
          'all_reused_input_hashes_match': True, 'compiler_module_count': len(inputs['compiler_sweep_modules']),
          'runtime': {'native_path': paths_before[0], 'native_sha256': sha(paths_before[0]), 'native_version': version.value,
                      'reference_version': torch.__version__, 'reference_cuda': torch.version.cuda,
                      'mapped_after_reference_import': paths_after},
          'recipe_hashes': {p.name:sha(p) for p in (ROOT / 'target/postcommit-e835f7f').iterdir() if p.suffix in ('.py','.sh')}}
(OUT / ('verification-' + sys.argv[1] + '.json')).write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps(report,indent=2))
