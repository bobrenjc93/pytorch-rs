"""Fresh clean-commit verification; reuse the existing test input manifests."""
import ctypes
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys

ROOT = Path.cwd().resolve()
OUT = ROOT / 'target/postcommit-f8f8244/capture'
COMMIT = 'f8f8244c2cfa81e255c1814d1bd101c5d846b03d'
sys.path.insert(0, str(ROOT / 'scripts'))
from evaluate_cuda_math import source_provenance

def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

assert subprocess.check_output(['git', 'rev-parse', 'HEAD'], text=True).strip() == COMMIT
assert subprocess.check_output(['git', 'status', '--porcelain'], text=True) == ''
assert Path(sys.prefix).resolve() == ROOT / '.venv' and not (ROOT / '.venv').is_symlink()
assert Path(sys.executable).resolve().is_relative_to(ROOT / 'target/uv-python')
assert sys.version_info[:3] == (3, 12, 14)
build_path = ROOT / 'target/postcommit-f8f8244/release/build-record.json'
build = json.loads(build_path.read_text())
assert build['clean_checkout'] and build['git_status_before_build'] == ''
assert build['commit'] == COMMIT
assert source_provenance()['source_sha256'] == build['source_sha256']
import torch_rs as native
assert Path(native.__file__).resolve().is_relative_to(ROOT / '.venv')
assert Path(native._C.__file__).resolve().is_relative_to(ROOT / '.venv')
assert sha(native._C.__file__) == build['extension_sha256']
assert sha(build['wheel_path']) == build['wheel_sha256']
for source in (ROOT / 'python/torch_rs').rglob('*.py'):
    installed = Path(native.__file__).parent / source.relative_to(ROOT / 'python/torch_rs')
    assert source.read_bytes() == installed.read_bytes(), source
inputs_path = ROOT / 'docs/diagnostics/compile-cuda-squeeze/inputs.json'
override_path = ROOT / 'docs/diagnostics/compile-cuda-squeeze/review-dynamic-consumers/verification-final.json'
inputs = json.loads(inputs_path.read_text())
overrides = json.loads(override_path.read_text())['changed_inputs']
for path, digest in (inputs['files'] | inputs['compiler_sweep_modules'] | overrides).items():
    assert sha(ROOT / path) == digest, path
assert set(inputs['compiler_sweep_modules']) == {str(p.relative_to(ROOT)) for p in ROOT.glob('tests/test_compile*.py')}
assert native.cuda.is_available()
def runtimes():
    return sorted({line.split()[-1] for line in Path('/proc/self/maps').read_text().splitlines() if '/libcudart.so' in line})
paths = runtimes()
assert paths == [os.environ['TORCH_RS_CUDART']], paths
runtime = ctypes.CDLL(paths[0])
version = ctypes.c_int()
assert runtime.cudaRuntimeGetVersion(ctypes.byref(version)) == 0
import torch
assert torch.__version__ == '2.13.0+cu130'
assert torch.cuda.device_count() == 1 and torch.cuda.get_device_name(0) == 'NVIDIA H100'
assert torch.version.cuda == '13.0' and version.value == 13000
assert runtimes() == paths
report = {**source_provenance(), 'git_status': '', 'measurement_kind': 'clean-code',
          'python': sys.version, 'executable': sys.executable,
          'interpreter_realpath': str(Path(sys.executable).resolve()),
          'interpreter_sha256': sha(Path(sys.executable).resolve()),
          'package_path': native.__file__, 'extension_path': native._C.__file__,
          'extension_sha256': sha(native._C.__file__), 'installed_python_matches_source': True,
          'build_record': 'build-record.json', 'build_record_sha256': sha(build_path),
          'input_manifest': '../inputs.json', 'input_manifest_sha256': sha(inputs_path),
          'input_override_manifest': '../review-dynamic-consumers/verification-final.json',
          'input_override_manifest_sha256': sha(override_path), 'all_reused_input_hashes_match': True,
          'compiler_module_count': len(inputs['compiler_sweep_modules']),
          'runtime': {'native_path': paths[0], 'native_sha256': sha(paths[0]), 'native_version': version.value,
                      'reference_version': torch.__version__, 'reference_cuda': torch.version.cuda}}
destination = OUT / ('verification-' + sys.argv[1] + '.json')
assert not destination.exists()
destination.write_text(json.dumps(report, indent=2) + '\n')
print(json.dumps(report, indent=2))
