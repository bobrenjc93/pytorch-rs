"""Non-scoring exact-main H100 baseline; no repository source edits.

Prepared while flatten review runs. This is not evidence until executed in a
fresh, source-bound environment after the current candidate is resolved.
"""
import hashlib
import json
import os
from pathlib import Path
import re
import subprocess
import sys

import numpy as np
import torch
import torch_rs as native

root = Path.cwd().resolve()
sys.path.insert(0, str(root))
from tests.test_cuda_contiguous import metadata, read_bits, write_bits


def squeeze_all(x):
    return x.squeeze()


def squeeze_keyword(x):
    return x.squeeze(dim=-1)


def squeeze_one(x):
    return x.squeeze(0)


def squeeze_tuple(x):
    return x.squeeze((0, -1))


def squeeze_list(x):
    return x.squeeze([0, -1])


def squeeze_variadic(x):
    return x.squeeze(0, -1)


def squeeze_empty_dims(x):
    return x.squeeze(())


def squeeze_transpose(x):
    return x.t().squeeze()


def affine_relu_squeeze(x, w):
    return (x @ w).relu().squeeze(0)


def compiled_gap(call):
    try:
        call()
    except NotImplementedError as error:
        assert type(error).__name__ == 'CompileTraceUnsupportedError'
        return {'type': type(error).__name__, 'message': str(error)}
    raise AssertionError('Expected unsupported native CUDA squeeze capture')


def outcome(call):
    try:
        result = call()
    except (TypeError, ValueError, IndexError, RuntimeError, NotImplementedError) as error:
        return {'success': False, 'type': type(error).__name__, 'message': str(error)}
    return {'success': True, 'shape': list(result.shape), 'stride': list(result.stride())}


head = subprocess.check_output(['git', 'rev-parse', 'HEAD'], text=True).strip()
assert len(sys.argv) == 2 and re.fullmatch(r'[0-9a-f]{40}', sys.argv[1]) and sys.argv[1] == head
assert subprocess.check_output(['git', 'rev-parse', 'main'], text=True).strip() == head
assert subprocess.check_output(['git', 'status', '--porcelain'], text=True) == ''
assert os.environ['CUDA_VISIBLE_DEVICES'] == '0'
assert torch.cuda.is_available() and torch.cuda.device_count() == 1
assert torch.cuda.get_device_name(0) == 'NVIDIA H100'
assert Path(sys.prefix).resolve() == root / '.venv'
assert Path(native._C.__file__).resolve().is_relative_to(root / '.venv')
rng = np.random.default_rng(198200)
cases = []
for label, fn, shapes in (
    ('leading_singleton', squeeze_all, [(1, 7)]),
    ('trailing_singleton', squeeze_all, [(7, 1)]),
    ('keyword_dimension', squeeze_keyword, [(7, 1)]),
    ('tuple_dimensions', squeeze_tuple, [(1, 1)]),
    ('list_dimensions', squeeze_list, [(1, 1)]),
    ('variadic_dimensions', squeeze_variadic, [(1, 1)]),
    ('unchanged_matrix', squeeze_all, [(3, 7)]),
    ('nonsingleton_dimension', squeeze_one, [(3, 7)]),
    ('empty_dimensions', squeeze_empty_dims, [(1, 7)]),
    ('vector', squeeze_all, [(7,)]),
    ('scalar', squeeze_all, [()]),
    ('scalar_dimension', squeeze_one, [()]),
    ('zero_sized', squeeze_all, [(1, 0)]),
    ('transposed_singleton', squeeze_transpose, [(1, 7)]),
    ('affine_relu', affine_relu_squeeze, [(1, 7), (7, 5)]),
):
    data = [np.asarray(rng.normal(size=shape), dtype=np.float32) for shape in shapes]
    # NumPy's empty arrays have zero strides; use equal canonical empty inputs.
    refs = [torch.tensor(value, device='cuda:0') if value.size else
            torch.empty(value.shape, dtype=torch.float32, device='cuda:0') for value in data]
    args = [native.tensor(value).to('cuda:0') for value in data]
    for actual_input, expected_input in zip(args, refs):
        assert metadata(actual_input) == metadata(expected_input), label
    expected, actual = fn(*refs), fn(*args)
    assert metadata(actual) == metadata(expected)
    np.testing.assert_allclose(np.asarray(actual.cpu().tolist()), expected.cpu().numpy(),
                               rtol=2e-5, atol=2e-6)
    assert actual is not args[0] and expected is not refs[0]
    aliases = fn is not affine_relu_squeeze
    assert (actual.data_ptr() == args[0].data_ptr()) == aliases
    assert (expected.data_ptr() == refs[0].data_ptr()) == aliases
    mutation_verified = False
    if aliases:
        np.testing.assert_array_equal(read_bits(actual), read_bits(expected))
        if args[0].numel():
            replacement = np.full(args[0].numel(), 17.25, dtype=np.float32).view(np.uint32)
            write_bits(args[0], replacement)
            write_bits(refs[0], replacement)
            np.testing.assert_array_equal(read_bits(actual), replacement)
            np.testing.assert_array_equal(read_bits(expected), replacement)
            mutation_verified = True
    torch._dynamo.reset()
    reference_compiled = torch.compile(fn, backend='eager', fullgraph=True, dynamic=False)
    for _ in range(2):
        result = reference_compiled(*refs)
        torch.testing.assert_close(result, expected, rtol=0, atol=0)
        assert metadata(result) == metadata(expected) and result is not refs[0]
    native_compiled = native.compile(fn, backend='eager', fullgraph=True, dynamic=False)
    cases.append({'name': label, 'function': fn.__name__, 'shapes': shapes,
                  'initial_input_sha256': [hashlib.sha256(value.tobytes()).hexdigest() for value in data],
                  'reference_eager_and_compiled_passed': True, 'native_eager_passed': True,
                  'input_address_matches': aliases, 'returns_original_input': False,
                  'storage_alias_mutation_verified': mutation_verified,
                  'empty_alias_not_proven_by_pointer': args[0].numel() == 0,
                  'native_compiled_cold': compiled_gap(lambda: native_compiled(*args)),
                  'native_compiled_repeat': compiled_gap(lambda: native_compiled(*args))})

values = np.arange(7, dtype=np.float32).reshape(1, 7)
a, b = native.tensor(values).to('cuda:0'), torch.tensor(values, device='cuda:0')
bindings = {}
for name, call in (
    ('positive_out_of_range', lambda x: x.squeeze(2)),
    ('negative_out_of_range', lambda x: x.squeeze(-3)),
    ('duplicate_dimensions', lambda x: x.squeeze((0, -2))),
    ('bool_dimension', lambda x: x.squeeze(True)),
    ('none_dimension', lambda x: x.squeeze(None)),
    ('huge_dimension', lambda x: x.squeeze(2**100)),
    ('huge_tuple_dimension', lambda x: x.squeeze((2**100,))),
    ('duplicate_binding', lambda x: x.squeeze(0, dim=1)),
    ('unexpected_keyword', lambda x: x.squeeze(other=0)),
):
    bindings[name] = {'reference': outcome(lambda: call(b)), 'native': outcome(lambda: call(a))}
    assert not bindings[name]['reference']['success']
    assert bindings[name]['reference']['type'] == bindings[name]['native']['type']

assert subprocess.check_output(['git', 'status', '--porcelain'], text=True) == ''
print(json.dumps({'verified_baseline_gap': True, 'commit': head,
                  'probe_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
                  'raw_bits_helper_sha256': hashlib.sha256((root / 'tests/test_cuda_contiguous.py').read_bytes()).hexdigest(),
                  'native_extension': native._C.__file__,
                  'native_extension_sha256': hashlib.sha256(Path(native._C.__file__).read_bytes()).hexdigest(),
                  'reference': torch.__version__, 'gpu': torch.cuda.get_device_name(0),
                  'cases': cases, 'bindings': bindings,
                  'reference_compile_backend': 'eager', 'cuda_performance_measured': False,
                  'physical_gpu_snapshot_after': subprocess.check_output(['nvidia-smi',
                      '--query-gpu=index,uuid,name,utilization.gpu,memory.used', '--format=csv,noheader'], text=True),
                  'gpu_snapshot_is_not_reservation': True, 'scores_changed': False}, indent=2))
