root=pathlib.Path.cwd();out=root/'target/postcommit-06a2496'
def run(label,command,env=None):

run('scoring-cache-preparation',[sys.executable,'-I','-B',str(out/'prepare_scoring.py')])

 ('scoring',[sys.executable,'-I','-B',str(out/'run_scoring.py'),'--output',str(out/'fixed-scoring.json')]),

]:

 run('fixed-scoring' if name=='scoring' else 'compiled-timings',command,env)
