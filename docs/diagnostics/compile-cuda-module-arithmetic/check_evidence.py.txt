"""Verify this development capture's hashes; no scoring or progress artifacts."""
import hashlib
import gzip
import json
from pathlib import Path
import subprocess
root = Path.cwd()
folder = root / 'docs/diagnostics/compile-cuda-module-arithmetic'
def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()
commands = json.loads((folder / 'commands.json').read_text())
states = json.loads((folder / 'input-manifests.json').read_text())
for name, entry in states['variants'].items():
    current = {k:v for k,v in states['base'].items() if k not in entry['removed']}
    current.update(entry['changed'])
    raw = json.dumps(current, sort_keys=True, indent=2) + '\n'
    assert hashlib.sha256(raw.encode()).hexdigest() == entry['sha256']
for name, record in commands.items():
    assert record['input_manifest_sha256'] == states['variants'][record['input_manifest']]['sha256']
    path = folder / record['published_log']
    raw = gzip.decompress(path.read_bytes()) if path.suffix == '.gz' else path.read_bytes()
    assert hashlib.sha256(raw).hexdigest() == record['log_sha256'], name
final = commands['compiler-final-source']
assert final['exit_code'] == 0
entry = states['variants'][final['input_manifest']]
inputs = {k:v for k,v in states['base'].items() if k not in entry['removed']}
inputs.update(entry['changed'])
assert all(sha(root / path) == digest for path, digest in inputs.items())
builds = json.loads((folder / 'builds.json').read_text())
build = builds['module-final-guarded']
assert all(sha(root / path) == digest for path, digest in build['source_files'].items())
assert sha(Path(build['installed_native'])) == build['installed_native_sha256']
assert sha(Path(build['wheel'])) == build['wheel_sha256']
allowed = {'python/torch_rs/_compile_bytecode.py','python/torch_rs/_compile_trace.py',
           'tests/test_compile_cuda_trailing_vector.py','ARCHITECTURE.md','FEATURES.md',
           'docs/README.md','docs/compile-cuda-add.md','docs/supported-surface.md'}
changed = set(subprocess.check_output(['git','diff','--name-only'],text=True).splitlines())
assert changed <= allowed, changed - allowed
assert subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip() == '64dd296ee65bff37d5d2868506de2e61aa5effb5'
print(json.dumps(dict(verified=True, command_count=len(commands), input_states=len(states['variants']),
    final_source_matches_compiler_run=True, installed_native_matches_release_wheel=True,
    original_base_commit_unchanged=True, tracked_changes_within_scope=True,
    clean_commit_capture_complete=False,
    clean_commit_capture_dependency='Burner implementation commit and delivery gates; this task forbids commits',
    artifact_sha256={str(p.relative_to(folder)):sha(p) for p in sorted(folder.rglob('*')) if p.is_file() and p.name != 'verification.json'}), indent=2))
