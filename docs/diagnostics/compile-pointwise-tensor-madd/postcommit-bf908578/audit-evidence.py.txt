"""Verify clean-source and raw-artifact identities; never rewrite measurements."""
from pathlib import Path
import gzip,hashlib,json,subprocess,zipfile,tarfile,os,sys,importlib.util
root=Path.cwd(); run=root/'target/tensor-madd/postcommit'
C='bf9085787aca29e523a7a38773f20f965167e734'; B='77aa16fc2d0cbd258f0fa309140dc75708cf2ee3'
sys.path.insert(0,str(root/'scripts'))
spec=importlib.util.spec_from_file_location('fixed_evaluator',root/'scripts/evaluate_torch_compile_default.py')
evaluator=importlib.util.module_from_spec(spec); spec.loader.exec_module(evaluator)
def sha(data): return hashlib.sha256(data).hexdigest()
def read(p): return json.loads(gzip.decompress(p.read_bytes()) if p.suffix=='.gz' else p.read_bytes())
def check_source(files,commit):
 for name,digest in files.items():
  assert sha(subprocess.check_output(['git','show',commit+':'+name]))==digest,name

def file_identity(record):
 p=Path(record['path']); assert p.resolve().is_relative_to(root)
 assert sha(p.read_bytes())==record['sha256'],str(p)

receipts=read(run/'commands.json')
assert len(receipts)==9 and all(c['returncode']==0 and c['sourceStatus']=='' for c in receipts)
assert 'Ran 10 tests' in (run/'two-device.log').read_text() and (run/'two-device.log').read_text().rstrip().endswith('OK')
historical=root/'docs/diagnostics/compile-pointwise-tensor-madd'
for name,digest in read(historical/'sha256.json').items(): assert sha((historical/name).read_bytes())==digest
build=read(run/'build/build.json'); assert build['passed'] and build['head']==C and build['status']==''
check_source(build['source'],C)
for key in ('wheel','extension'): file_identity(build[key])
summary={'candidate':C,'main':B,'historicalManifestFiles':len(read(historical/'sha256.json')),'diagnostic':[],'fixed':{}}
for i,kind in enumerate(('native','reference','reference','native')):
 directory=run/f'paired-{i}-{kind}'; data=read(directory/'report.json.gz')
 assert data['passed'] and data['source']['head']==C and data['source']['status']==''
 assert data['source']['files']==build['source']
 assert data['scriptSha256']==sha((historical/'capture.py').read_bytes())
 assert data['helperSha256']==sha((root/'docs/diagnostics/compile-pointwise-positional/warm-dispatch/gpu-dispatch.py').read_bytes())
 assert data['build']['wheel']==build['wheel'] and data['build']['extension']==build['extension']
 states=[s for c in data['cases'] for s in c['states']]; assert len(states)==42
 for s in states:
  assert len(s['calls'])==23
  assert [c['phase'] for c in s['calls']]==['cold']+['warmup']*5+['sample']*17
  if kind=='native':
   assert s['nvrtc']==[13,0]
   for call in s['calls']:
    for key,ext in (('kernelSourceSha256','.cu'),('kernelPtxSha256','.ptx')):
     code=(directory/(call[key]+ext)).read_bytes(); assert sha(code)==call[key]
     assert code.count(b'fmaf(' if ext=='.cu' else b'fma.rn.f32')==1
 summary['diagnostic'].append({'implementation':kind,'states':len(states),'calls':sum(len(s['calls']) for s in states),'passed':True})
for a,b in ((0,1),(2,3)):
 data=read(run/f'comparison-{a}-{b}.json'); assert data['passed'] and len(data['comparisons'])==42 and not data['failures']
 for name,digest in zip(data['reports'],data['reportSha256']): assert sha(Path(name).read_bytes())==digest
for label,r,commit in (('candidate',root,C),('main',run/'main',B)):
 reports=list((r/'target/default-compile-eval').glob('run-*/report.json')); assert len(reports)==1
 data=read(reports[0]); assert data['valid'] and not data['diagnostic'],data.get('error')
 assert data['provenance']['commit']==commit and data['provenance']['working_tree_changes']==''
 check_source(data['provenance']['files'],commit)
 assert len(data['cells'])==112 and sum(c['device']=='cuda' for c in data['cells'])==56
 assert len(data['workers'])==6 and data['warmups']==5 and data['samples']==17
 assert evaluator.aggregate(data['cells'],'coverage')==data['scores']['coverage']
 assert evaluator.aggregate([c for c in data['cells'] if c['device']=='cuda'],'cuda-perf')==data['scores']['cuda-perf']
 file_identity(data['wheel'])
 for worker in data['workers']:
  art=worker['artifacts']
  for key in ('output','log'): assert sha(Path(art[key]).read_bytes())==art[key+'_sha256']
  raw=read(Path(art['output'])); assert len(raw['cases'])==28
  env=raw['environment']; assert Path(env['executable']).resolve().is_relative_to(r)
  if raw['implementation']=='reference':
   assert all(c['status']=='passed' for c in raw['cases'].values())
   for record in env['reference_provenance']['origins'].values(): file_identity(record)
  else:
   file_identity({'path':env['native_extension'],'sha256':env['native_sha256']})
  for case in raw['cases'].values():
   if case['status']=='passed':
    assert len(case['variants'])==2
    for variant in case['variants']:
     assert len(variant['samples_ms'])==17
 with zipfile.ZipFile(data['wheel']['path']) as wheel:
  site=r/'target/default-compile-eval/venv/lib/python3.12/site-packages'
  for name in wheel.namelist():
   if name.startswith('torch_rs/') and name.endswith(('.py','.so')): assert (site/name).read_bytes()==wheel.read(name)
 summary['fixed'][label]={'commit':commit,'coverageCells':len(data['cells']),'cudaCells':56,'scores':data['scores'],'passedCells':sum(c['passed'] for c in data['cells']),'passedCudaCells':sum(c['passed'] for c in data['cells'] if c['device']=='cuda'),'valid':True}
print(json.dumps(summary,indent=2))
(run/'audit.json').write_text(json.dumps(summary,indent=2)+'\n')
