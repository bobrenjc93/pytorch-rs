"""Copy raw post-commit evidence without editing report contents."""
from pathlib import Path
import hashlib,json,gzip,shutil,tarfile,datetime
root=Path.cwd(); run=root/'target/tensor-madd/postcommit'
out=root/'docs/diagnostics/compile-pointwise-tensor-madd/postcommit-bf908578'
out.mkdir(exist_ok=False)
def copy(source,destination):
 destination.parent.mkdir(parents=True,exist_ok=True); shutil.copyfile(source,destination)
 assert source.read_bytes()==destination.read_bytes()
def tar(destination,base,files):
 with tarfile.open(destination,'w:gz') as archive:
  for source in sorted(files): archive.add(source,arcname=source.relative_to(base),recursive=False)
for name in ('commands.json','sequence.log','prepare.log','build-driver.log','run-evidence.py','prepare-environments.py','archive-evidence.py','audit-evidence.py','audit.json','audit.log','ownership.json','inspection-notes.txt'):
 copy(run/name,out/(name+'.txt' if name.endswith('.py') else name))
copy(root/'target/tensor-madd/env.sh',out/'environment.sh.txt')
for source in sorted(run.glob('*.log')):
 copy(source,out/'logs'/source.name)
for name in ('build.json','build.log','install.log','source.tar.gz'):
 copy(run/'build'/name,out/'build'/name)
for source in sorted((run/'build/wheels').glob('*.whl')):
 copy(source,out/'build'/source.name)
for i,kind in enumerate(('native','reference','reference','native')):
 name=f'paired-{i}-{kind}'; directory=run/name
 copy(directory/'report.json.gz',out/(name+'.json.gz'))
 tar(out/(name+'.tar.gz'),directory,[p for p in directory.iterdir() if p.is_file() and p.suffix in ('.cu','.ptx')])
for name in ('comparison-0-1.json','comparison-2-3.json'):
 copy(run/name,out/name)
retention=[]
for label,r in (('candidate',root),('main',run/'main')):
 reports=list((r/'target/default-compile-eval').glob('run-*/report.json'))
 assert len(reports)==1, label
 for report in reports:
  data=json.loads(report.read_text()); destination=out/'fixed'/label/report.parent.name
  staged=root/'target/default-compile-eval'/report.parent.name
  if label=='main': staged.mkdir(exist_ok=False)
  for source in sorted(report.parent.iterdir()):
   if not source.is_file(): continue
   retained=staged/source.name
   if label=='main': copy(source,retained)
   if source.suffix!='.gz': copy(source,destination/source.name)
   retention.append({'capture':label,'source':str(source),'observerSource':str(retained),
                     'bytes':source.stat().st_size,'sha256':hashlib.sha256(source.read_bytes()).hexdigest()})
  if 'wheel' in data:
   source=Path(data['wheel']['path']); copy(source,out/'fixed'/label/'wheel'/source.name)
(out/'raw-retention.json').write_text(json.dumps({'stagedAt':datetime.datetime.now(datetime.timezone.utc).isoformat(),'mechanism':'Installed operator pre-cleanup observer; raw main-export files staged byte-for-byte under the canonical observed report root. No evaluator or observer changes.','files':retention},indent=2)+'\n')
print(out)
