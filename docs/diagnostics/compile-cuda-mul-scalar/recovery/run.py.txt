import json, os, pathlib, subprocess, sys, time
out = pathlib.Path(__file__).resolve().parent
name, *command = sys.argv[1:]
assert name and command
log = out / (name + '.log')
receipt = out / (name + '.receipt.json')
assert not log.exists() and not receipt.exists(), name
record = {'name': name, 'command': command, 'cwd': os.getcwd(), 'started_utc': time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime()),
          'environment': {k: os.environ.get(k) for k in ('CUDA_VISIBLE_DEVICES','PYO3_PYTHON','VIRTUAL_ENV','TORCH_RS_VERIFY_VIRTUALENV','CARGO_HOME','CARGO_TARGET_DIR','TMPDIR','UV_CACHE_DIR','UV_PYTHON_INSTALL_DIR','XDG_CACHE_HOME','CUDA_CACHE_PATH','TORCHINDUCTOR_CACHE_DIR','TRITON_CACHE_DIR','PYTHONPYCACHEPREFIX','PYTHONHASHSEED')}}
receipt.write_text(json.dumps(record, indent=2)+'\n')
start = time.monotonic()
with log.open('w') as stream:
    result = subprocess.run(command, stdout=stream, stderr=subprocess.STDOUT)
record.update(exit_code=result.returncode, elapsed_seconds=time.monotonic()-start)
receipt.write_text(json.dumps(record, indent=2)+'\n')
print(name, result.returncode, flush=True)
sys.exit(result.returncode)
