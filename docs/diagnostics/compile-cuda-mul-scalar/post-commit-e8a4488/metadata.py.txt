import hashlib, importlib.metadata as md, json, os, pathlib, platform, subprocess, sys, sysconfig, time, zipfile
root = pathlib.Path.cwd()
sys.path.insert(0, str(root))
from scripts.diagnose_compile_cuda_neg_add import environment
out = root / 'target/post-commit-e8a4488'
version = sys.argv[1]
expected_commit = 'e8a4488aa0751348a8e91f744b2d76e934576ba9'
record = environment('0')
assert record['base_commit'] == expected_commit
assert record['worktree_status'] == '', record['worktree_status']
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
paths = subprocess.check_output(['git', 'ls-files', 'src', 'python', 'scripts', 'tests', 'Cargo.toml', 'Cargo.lock', 'pyproject.toml', 'uv.lock']).decode().splitlines()
record['source_sha256'].update({p: sha(root / p) for p in paths})
record['measured_code_commit'] = record.pop('base_commit')
record['recorded_utc'] = time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime())
record['source_root'] = str(root)
record['python'] = {'version':sys.version, 'executable':sys.executable, 'base_executable':sys._base_executable, 'build':platform.python_build(), 'compiler':platform.python_compiler(), 'configuration':{k:sysconfig.get_config_var(k) for k in ('CONFIG_ARGS','CC','CFLAGS','SOABI','Py_DEBUG','Py_GIL_DISABLED')}}
record['packages'] = {d.metadata['Name']:d.version for d in md.distributions()}
record['toolchain'] = {tool:subprocess.check_output(args, text=True).strip() for tool,args in [('rustc',['rustc','-Vv']),('cargo',['cargo','-V']),('nvcc',['nvcc','--version']),('uv',['uv','--version'])]}
record['build'] = json.loads((out / f'build-{version}.receipt.json').read_text())
record['dependency_setup'] = json.loads((out / f'sync-{version}.receipt.json').read_text())
record['dependency_setup']['project_environment'] = sys.prefix
record['cache_policy'] = 'Fresh virtualenvs, Cargo targets and CUDA/compiler caches for this post-commit run; worktree-local uv/Cargo download caches and managed Python installation reused.'
record['build_profile'] = 'release; extension-module, abi3-py310; thin LTO and codegen-units=1 from committed Cargo.toml'
wheel, = (out / f'wheels{version}').glob('*.whl')
record['wheel'] = str(wheel)
record['wheel_sha256'] = sha(wheel)
with zipfile.ZipFile(wheel) as zipped:
    native, = [name for name in zipped.namelist() if name.endswith('.so')]
    assert hashlib.sha256(zipped.read(native)).hexdigest() == record['native_extension_sha256']
    for p in (root / 'python/torch_rs').rglob('*.py'):
        assert zipped.read(str(p.relative_to(root / 'python'))) == p.read_bytes(), p
(out / f'build-provenance-{version}.json').write_text(json.dumps(record, indent=2)+'\n')
print('Verified clean commit, wheel/source/native equality and installed import provenance:', expected_commit)
