import datetime, json, os, pathlib, subprocess, sys, time
root = pathlib.Path(__file__).resolve().parent
label, *command = sys.argv[1:]
record = {'command': command, 'cwd': os.getcwd(), 'started_utc': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'environment': {k:v for k,v in os.environ.items() if k in ('CUDA_VISIBLE_DEVICES','TORCH_RS_CUDART','CARGO_HOME','CARGO_TARGET_DIR','UV_CACHE_DIR','UV_PROJECT_ENVIRONMENT','PYO3_PYTHON','VIRTUAL_ENV','CUDA_CACHE_PATH','TRITON_CACHE_DIR','TORCHINDUCTOR_CACHE_DIR','RUSTUP_TOOLCHAIN','TMPDIR','PYTHONPYCACHEPREFIX')}}
for label_git, args in [('commit',['rev-parse','HEAD']),('worktree_status',['status','--porcelain'])]:
    record[label_git] = subprocess.check_output(['git',*args],text=True).strip()
t = time.monotonic()
with (root/'logs'/f'{label}.log').open('w') as out:
    result = subprocess.run(command,stdout=out,stderr=subprocess.STDOUT)
record.update(exit_code=result.returncode,elapsed_seconds=time.monotonic()-t,finished_utc=datetime.datetime.now(datetime.timezone.utc).isoformat())
(root/'logs'/f'{label}.receipt.json').write_text(json.dumps(record,indent=2)+'\n')
print(label, result.returncode, record['elapsed_seconds'],flush=True)
sys.exit(result.returncode)
