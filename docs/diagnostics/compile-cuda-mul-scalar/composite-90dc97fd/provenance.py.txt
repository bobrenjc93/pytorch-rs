import datetime, hashlib, importlib.metadata as md, json, os, pathlib, platform, subprocess, sys, sysconfig, zipfile
root = pathlib.Path.cwd()
sys.path.insert(0, str(root))
from scripts.diagnose_compile_cuda_neg_add import environment
out = root.parent
version = sys.argv[1]
record = environment('0')
commit = subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip()
assert record['base_commit'] == commit and record['worktree_status'] == ''
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
paths = subprocess.check_output(['git','ls-files','src','python','scripts','tests','Cargo.toml','Cargo.lock','pyproject.toml','uv.lock','rust-toolchain.toml']).decode().splitlines()
record['source_sha256'].update({p:sha(root/p) for p in paths})
record.update(measured_code_commit=commit, recorded_utc=datetime.datetime.now(datetime.timezone.utc).isoformat(),source_root=str(root))
record['python'] = dict(version=sys.version, executable=sys.executable,base_executable=sys._base_executable,build=platform.python_build(),compiler=platform.python_compiler(),configuration={k:sysconfig.get_config_var(k) for k in ('CONFIG_ARGS','CC','CFLAGS','SOABI','Py_DEBUG','Py_GIL_DISABLED')})
record['python']['base_executable_sha256'] = sha(pathlib.Path(sys._base_executable))
record['packages'] = {d.metadata['Name']:d.version for d in md.distributions()}
record['toolchain'] = {tool:subprocess.check_output(args,text=True).strip() for tool,args in [('rustc',['rustc','-Vv']),('cargo',['cargo','-V']),('nvcc',['nvcc','--version']),('uv',['uv','--version'])]}
name = 'managed312' if version == '312' else '314'
record['build'] = json.loads((out/'logs'/f'build-{name}.receipt.json' if version == '312' else out/'logs'/'build314.receipt.json').read_text())
record['dependency_setup'] = json.loads((out/'logs'/('sync312-managed.receipt.json' if version == '312' else 'sync314.receipt.json')).read_text())
record['cache_policy'] = 'Fresh virtualenvs and per-interpreter Cargo release targets. Python 3.12.13 base was installed locally; Python 3.14 base was reused read-only from the host. Worktree-local dependency downloads reused after initial cold setup; CUDA/compiler caches reused across correctness checks. No timing inferred from diagnostics.'
record['build_profile'] = 'release; extension-module, abi3-py310; thin LTO; codegen-units=1; locked dependencies'
wheel, = (out/('wheels-managed312' if version == '312' else 'wheels314')).glob('*.whl')
record['wheel'] = str(wheel)
record['wheel_sha256'] = sha(wheel)
with zipfile.ZipFile(wheel) as zipped:
    native, = [name for name in zipped.namelist() if name.endswith('.so')]
    assert hashlib.sha256(zipped.read(native)).hexdigest() == record['native_extension_sha256']
    for p in (root/'python/torch_rs').rglob('*.py'):
        assert zipped.read(str(p.relative_to(root/'python'))) == p.read_bytes(), p
for path in (record['native_extension'],record['environment']['executable'],record['environment']['package'],record['environment']['native_runtime_library'],record['wheel']):
    pathlib.Path(path).absolute().relative_to(out)
    assert pathlib.Path(path).exists(),path
(out/f'provenance-{version}.json').write_text(json.dumps(record,indent=2)+'\n')
print('Verified clean composite commit, local interpreter/runtime, committed source, wheel and installed native:',commit)
