import ctypes, importlib.metadata, json, pathlib, sys
import torch, torch_rs
from torch_rs._cuda_public_storage import _candidate_libraries
lib = ctypes.CDLL(_candidate_libraries()[0])
runtime = ctypes.c_int(); driver = ctypes.c_int()
assert lib.cudaRuntimeGetVersion(ctypes.byref(runtime)) == 0
assert lib.cudaDriverGetVersion(ctypes.byref(driver)) == 0
print(json.dumps({'python':sys.version, 'python_executable':sys.executable,
'interpreter':str(pathlib.Path(sys.executable).resolve()),
'native_import':torch_rs.__file__, 'extension':torch_rs._C.__file__,
'reference':torch.__version__, 'reference_cuda':torch.version.cuda,
'native_runtime_library':lib._name, 'native_runtime':runtime.value,'native_driver':driver.value,
'gpu':torch.cuda.get_device_name(), 'capability':torch.cuda.get_device_capability(),
'visible_count':torch.cuda.device_count()}, indent=2))
