"""Publish compact development receipts using the existing capture contracts."""
import difflib
import hashlib
import json
from pathlib import Path
import shutil
import subprocess
import sys
ROOT = Path.cwd()
OUT = ROOT / 'docs/diagnostics/compile-cuda-transpose'
WORK = ROOT / 'target/transpose'
sys.path.insert(0, str(ROOT / 'scripts'))
from evaluate_cuda_math import source_provenance, sha256

def dump(path, obj):
    path.write_text(json.dumps(obj, indent=2, sort_keys=True) + '\n')

final = json.loads((WORK/'verified-build/build-record.json').read_text())
assert all(final[k] == v for k,v in source_provenance().items())
assert not final['clean_checkout'] and final['measurement_kind'] == 'precommit-diagnostic'
base = json.loads((WORK/'baseline/build-record.json').read_text())
assert base['clean_checkout'] and base['commit'] == '1ff6e6b3a57c69fec9f8b1292bf6ed30e00dec0b'
for build in ('baseline','development','final-build','verified-build'):
    dest = OUT/build
    dest.mkdir(exist_ok=True)
    for name in ('build-record.json','commands.json','build.log','install.log'):
        shutil.copyfile(WORK/build/name, dest/name)

receipts = {}
for p in (WORK/'evidence').glob('*.receipt.json'):
    receipt = json.loads(p.read_text())
    name = p.name.removesuffix('.receipt.json')
    assert receipt['source_before'] == receipt['source_after']
    assert receipt['inputs_unchanged'] and receipt['git_status_before'] == receipt['git_status_after']
    assert receipt['log_sha256'] == sha256(WORK/'evidence'/f'{name}.log')
    assert receipt['exit_status'] == (1 if name in ('transpose-first', 'transpose-final') else 0), name
    if name in ('transpose-verified','compiler-regressions','two-device','no-device','rust-final','cpu-final','imports','docs-verified'):
        assert receipt['source_before'] == source_provenance(), name
        assert set(receipt['native_before'].values()) == {final['extension_sha256']}, name
        assert receipt['native_before'] == receipt['native_after'], name
    receipts[name] = receipt
    shutil.copyfile(p, OUT/p.name)
    shutil.copyfile(WORK/'evidence'/f'{name}.log', OUT/f'{name}.log')

manifests = {sha256(p): json.loads(p.read_text()) for p in (WORK/'evidence').glob('inputs-*.json')}
final_hash = receipts['transpose-verified']['measured_inputs_sha256']
manifest = manifests[final_hash]
dump(OUT/'measured-inputs.json',manifest)
assert sha256(OUT/'measured-inputs.json') == final_hash
index = {}
for sha, measured in manifests.items():
    if sha == final_hash:
        index[sha] = {'file': 'measured-inputs.json'}
        continue
    delta = {p: value for p,value in measured.items() if manifest.get(p) != value}
    removed = sorted(set(manifest)-set(measured))
    name = 'inputs-delta-'+sha[:12]+'.json'
    dump(OUT/name, {'replace': delta, 'remove': removed})
    reconstructed = {k:v for k,v in manifest.items() if k not in removed} | delta
    serialized = json.dumps(reconstructed, indent=2, sort_keys=True)+'\n'
    assert hashlib.sha256(serialized.encode()).hexdigest() == sha
    index[sha] = {'base': 'measured-inputs.json', 'delta': name}
dump(OUT/'manifest-index.json',index)

patch = []
for path, saved in [('python/torch_rs/_compile_trace.py','trace-first.py.txt'),
                    ('tests/test_compile_cuda_transpose.py','transpose-first.py.txt'),
                    ('python/torch_rs/__init__.py','init-before-guard.py.txt')]:
    assert sha256(WORK/'evidence'/saved) == manifests[receipts['transpose-first']['measured_inputs_sha256']][path]
    patch.extend(difflib.unified_diff((ROOT/path).read_text().splitlines(keepends=True),
        (WORK/'evidence'/saved).read_text().splitlines(keepends=True), fromfile='a/'+path, tofile='b/'+path))
(OUT/'first-attempt.patch').write_text(''.join(patch))
patch = []
for path, saved in [('python/torch_rs/__init__.py','init-before-guard.py.txt'),
                    ('tests/test_compile_cuda_transpose.py','transpose-before-guard.py.txt')]:
    assert sha256(WORK/'evidence'/saved) == manifests[receipts['transpose-final']['measured_inputs_sha256']][path]
    patch.extend(difflib.unified_diff((ROOT/path).read_text().splitlines(keepends=True),
        (WORK/'evidence'/saved).read_text().splitlines(keepends=True), fromfile='a/'+path, tofile='b/'+path))
(OUT/'guard-failure.patch').write_text(''.join(patch))
for source,name in [('capture.py','capture.py.txt'),('env.sh','environment.sh.txt'),
                    ('baseline.py','baseline.py.txt'),('preflight.py','preflight.py.txt'),
                    ('publish.py','publish.py.txt'), ('setup-commands.txt','setup-commands.txt'), ('docs-example.py','docs-example.py.txt')]:
    shutil.copyfile(WORK/source,OUT/name)
shutil.copyfile(WORK/'evidence/setup.log',OUT/'setup.log')

import torch_rs
installed = Path(torch_rs.__file__).parent
source = ROOT/'python/torch_rs'
for path in source.rglob('*.py'):
    assert sha256(path) == sha256(installed/path.relative_to(source)), path
assert sha256(torch_rs._C.__file__) == final['extension_sha256']
assert sha256(source/Path(torch_rs._C.__file__).name) == final['extension_sha256']
changed = subprocess.check_output(['git','diff','--name-only','HEAD'],text=True).splitlines()
allowed = {'ARCHITECTURE.md','FEATURES.md','docs/README.md','docs/compile-cuda-t.md',
    'docs/supported-surface.md','python/torch_rs/_compile_trace.py','python/torch_rs/_compile_bytecode.py','python/torch_rs/__init__.py',
    'src/python_compile_cuda_graph.rs','src/tensor_cuda_graph.rs','tests/test_compile_cuda_t.py'}
assert set(changed) <= allowed, changed
assert subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip() == base['commit']
dump(OUT/'audit.json', {'source': source_provenance(), 'final_extension_sha256': final['extension_sha256'],
    'installed_python_matches_source':True, 'installed_native_matches_build':True,
    'baseline_clean_build_verified':True, 'candidate_clean_commit_capture':'pending Burner delivery commit; agent cannot commit',
    'manifest_count':len(manifests),'full_manifests_published':1,
    'checks':{name:record['exit_status'] for name,record in receipts.items()},
    'tracked_changed_paths':changed,'managed_artifacts_and_evaluators_unchanged':True,
    'changed_docs_sha256':{p:sha256(ROOT/p) for p in changed if p.endswith('.md')}})
dump(OUT/'inventory.json', {str(p.relative_to(OUT)):sha256(p) for p in OUT.rglob('*')
    if p.is_file() and p.name != 'inventory.json'})
print('Published',len(receipts),'receipts; verified source/build/imports, deduplicated manifests, retained first failure.')
