import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
import time
from datetime import datetime, timezone
root = Path.cwd()
sys.path.insert(0, str(root / 'scripts'))
from evaluate_cuda_math import source_provenance
name, *command = sys.argv[1:]
output = root / 'target/cuda-contiguous-evidence'
log = output / (name + '.log')
receipt = output / (name + '.receipt.json')
assert not log.exists() and not receipt.exists()
def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()
before = source_provenance()
started = datetime.now(timezone.utc).isoformat()
tick = time.perf_counter()
with log.open('w') as stream:
    result = subprocess.run(command, stdout=stream, stderr=subprocess.STDOUT)
extensions = list((root / '.venv/lib/python3.12/site-packages/torch_rs').glob('*.so'))
record = dict(command=command, cwd=str(root), started_at=started,
              ended_at=datetime.now(timezone.utc).isoformat(), wall_seconds=time.perf_counter()-tick,
              exit_status=result.returncode, log_sha256=sha(log), source_before=before,
              source_after=source_provenance(),
              extensions={str(p): sha(p) for p in extensions},
              environment={k: os.environ.get(k) for k in ('CUDA_VISIBLE_DEVICES', 'TORCH_RS_CUDART',
                  'CARGO_HOME', 'CARGO_TARGET_DIR', 'UV_CACHE_DIR', 'TMPDIR', 'XDG_CACHE_HOME',
                  'CUDA_CACHE_PATH', 'PYO3_PYTHON', 'VIRTUAL_ENV')})
receipt.write_text(json.dumps(record, indent=2)+'\n')
print(log.read_text())
sys.exit(result.returncode)
