from dataclasses import replace
from unittest.mock import patch
import torch_rs as native
from torch_rs import _compile_trace as trace
from tests.test_compile_cuda_contiguous import composed
from tests.test_compile_cuda_boundary import compile_with_cache
x=native.ones((3,7)).to('cuda:0').t()
f,cache=compile_with_cache(composed,dynamic=True)
f(x)
key,graph=next(iter(cache.graphs.items()))
last=graph.operations[-1]
cache.graphs[key]=replace(graph,operations=(*graph.operations[:-1],replace(last,metadata=replace(last.metadata,shape=(8,3)))))
with patch.object(trace._native,'_compile_trace_cuda_graph',side_effect=AssertionError('malformed dynamic declaration reached native execution')):
    try: f(x)
    except (ValueError,NotImplementedError): pass
