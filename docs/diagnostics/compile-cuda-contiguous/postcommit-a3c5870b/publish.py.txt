"""Publish byte-identical clean captures after all commands and audits finish."""
import hashlib, json, re, shutil, subprocess
from pathlib import Path
ROOT=Path.cwd().resolve()
OUT=ROOT/'target/contiguous-postcommit-a3c5870b'
DEST=ROOT/'docs/diagnostics/compile-cuda-contiguous/postcommit-a3c5870b'
assert subprocess.check_output(['git','status','--porcelain'],text=True)==''
audit=json.loads((OUT/'audit.json').read_text())
assert audit['status']=='passed'
receipt=json.loads((OUT/'evidence-audit.receipt.json').read_text())
assert receipt['exit_status']==0 and receipt['git_status_before']==receipt['git_status_after']==''
assert not DEST.exists()
DEST.mkdir()
for path in OUT.iterdir():
    if path.is_file():
        name=path.name+'.txt' if path.suffix in ('.py','.sh') else path.name
        shutil.copyfile(path,DEST/name)
        assert path.read_bytes()==(DEST/name).read_bytes()
(DEST/'release').mkdir()
for name in ('build-record.json','commands.json','build.log','install.log'):
    shutil.copyfile(OUT/'release'/name,DEST/'release'/name)
    assert (OUT/'release'/name).read_bytes()==(DEST/'release'/name).read_bytes()
inventory={str(p.relative_to(DEST)):hashlib.sha256(p.read_bytes()).hexdigest() for p in DEST.rglob('*') if p.is_file()}
(DEST/'inventory.json').write_text(json.dumps(inventory,indent=2,sort_keys=True)+'\n')

def result(name):
    log=(OUT/f'{name}.log').read_text()
    ran=int(re.search(r'Ran (\d+) tests?',log).group(1))
    skipped=re.search(r'OK \(skipped=(\d+)\)',log)
    skipped=int(skipped.group(1)) if skipped else 0
    assert '\nOK' in log
    return f'{ran-skipped} passed'+(f'; {skipped} hardware skip'+('s' if skipped!=1 else '') if skipped else '')
