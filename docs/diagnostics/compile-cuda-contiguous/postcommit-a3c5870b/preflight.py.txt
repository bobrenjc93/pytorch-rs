import ctypes, os, sys, pathlib, json, subprocess
import numpy, torch, torch_rs
root = pathlib.Path.cwd()
assert pathlib.Path(sys.prefix).resolve() == root/'.venv'
for m in (numpy, torch, torch_rs, torch_rs._C):
    path = pathlib.Path(m.__file__).resolve()
    assert path.is_relative_to(root/'.venv'), path
    print(m.__name__, getattr(m,'__version__',''), path)
print('interpreter',sys.executable,sys.version)
print('reference CUDA',torch.version.cuda,torch.cuda.device_count(),torch.cuda.get_device_name(0))
lib = ctypes.CDLL(os.environ['TORCH_RS_CUDART'])
v=ctypes.c_int()
assert lib.cudaRuntimeGetVersion(ctypes.byref(v)) == 0
print('native runtime',os.environ['TORCH_RS_CUDART'],v.value)
for command in (['nvidia-smi','--query-gpu=index,uuid,name,driver_version,compute_cap,utilization.gpu,memory.used,memory.total','--format=csv'],['nvcc','--version'],['rustc','-Vv'],['cargo','-V']):
    print(subprocess.check_output(command,text=True))
print('nvcc unused: native CUDA kernels use driver-JIT embedded PTX')
