"""Archive byte-identical repair diagnostics, keeping earlier captures intact."""
import hashlib, json, re, shutil
from pathlib import Path
ROOT = Path.cwd().resolve()
OUT = ROOT / 'target/contiguous-review-fixes'
DEST = ROOT / 'docs/diagnostics/compile-cuda-contiguous/review-fixes'
audit = json.loads((OUT / 'audit.json').read_text())
assert audit['status'] == 'passed'
assert json.loads((OUT / 'evidence-audit.receipt.json').read_text())['exit_status'] == 0
assert not DEST.exists()
DEST.mkdir()
manifest_names = {}
for path in OUT.iterdir():
    if not path.is_file():
        continue
    if path.name.endswith('.inputs.json'):
        digest = hashlib.sha256(path.read_bytes()).hexdigest()
        name = f'inputs/{digest}.json'
        manifest_names[path.name] = name
        (DEST / 'inputs').mkdir(exist_ok=True)
    else:
        name = path.name + '.txt' if path.suffix in ('.py', '.sh') else path.name
    shutil.copyfile(path, DEST / name)
    assert path.read_bytes() == (DEST / name).read_bytes()
(DEST / 'input-manifest-index.json').write_text(json.dumps(manifest_names, indent=2) + '\n')
for directory in ('release', 'release-final'):
    (DEST / directory).mkdir()
    for name in ('build-record.json', 'commands.json', 'build.log', 'install.log'):
        shutil.copyfile(OUT / directory / name, DEST / directory / name)
        assert (OUT / directory / name).read_bytes() == (DEST / directory / name).read_bytes()
inventory = {str(p.relative_to(DEST)): hashlib.sha256(p.read_bytes()).hexdigest()
             for p in DEST.rglob('*') if p.is_file()}
(DEST / 'inventory.json').write_text(json.dumps(inventory, indent=2, sort_keys=True) + '\n')

def result(name):
    log = (OUT / f'{name}.log').read_text()
    assert '\nOK' in log
    ran = int(re.search(r'Ran (\d+) tests?', log).group(1))
    match = re.search(r'OK \(skipped=(\d+)\)', log)
    skips = int(match.group(1)) if match else 0
    return f'{ran - skips} passed' + (f'; {skips} hardware skips' if skips else '')
