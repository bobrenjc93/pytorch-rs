import datetime,hashlib,json,pathlib,subprocess,sys,time
ROOT=pathlib.Path.cwd();OUT=ROOT/'target/composite-postcommit-d94daecd';BASE=OUT/'baseline'
def read(p):return json.loads(p.read_text())
def save(p,v):p.write_text(json.dumps(v,indent=2)+'\n')
def stamp():return datetime.datetime.now(datetime.timezone.utc).isoformat()
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()

receipts=[]
def run(name,root,command,log,env=None):
 before=subprocess.check_output(['git','status','--porcelain'],cwd=root,text=True);assert not before
 receipt=dict(command=command,cwd=str(root),started_at=stamp(),status_before=before,kind='read-only post-capture verification')
 tick=time.perf_counter()
 with log.open('x') as stream:r=subprocess.run(command,cwd=root,env=env,stdout=stream,stderr=subprocess.STDOUT)
 receipt.update(ended_at=stamp(),wall_seconds=time.perf_counter()-tick,exit_status=r.returncode,status_after=subprocess.check_output(['git','status','--porcelain'],cwd=root,text=True),log=str(log),log_sha256=sha(log))
 save(log.with_suffix('.receipt.json'),receipt);receipts.append(receipt);save(OUT/'verification-commands.json',receipts);print(name,r.returncode,flush=True);r.check_returncode()
