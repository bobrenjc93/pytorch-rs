"""Read-only accounting of all eight unchanged diagnostic reports."""
import json,math,pathlib,statistics,sys
ROOT=pathlib.Path.cwd();OUT=ROOT/'target/composite-postcommit-d94daecd'
SIDES={'baseline':OUT/'baseline/target/clean-d94daecd','candidate':ROOT/'target/clean-d94daecd'}
COMMITS={'baseline':'b7936239ceb7d7713a15ecd8a2c82c4bddd6d214','candidate':'d94daecdd9c086ac23afe690a52df11cbaca4f8a'}
sys.path.insert(0,str(ROOT/'scripts'))
from diagnose_compile_cuda_matmul import cells,aggregate
from evaluate_cuda_math import sha256

def geom(values):return math.exp(statistics.mean(map(math.log,values))) if values and all(v>0 for v in values) else 0.
count=0;pairs=[]
for name,seed in [('primary',798431),('held-481723',481723),('held-926051',926051),('primary-repeat',798431)]:
 reports=[]
 for side,folder in SIDES.items():
  path=folder/(name+'.json');report=json.loads(path.read_text())
  assert report['seed']==seed and report['measurement_kind']=='clean-commit' and not report['git_status']
  assert report['build_record']['record']['clean_checkout']
  assert report['source']['commit']==COMMITS[side] and report['source']==report['source_after']
  assert len(report['cases'])==len(cells(seed))==12
  for cell,expected in zip(report['cases'],cells(seed)):
   for key in ['shape_mkn','offset','program']:
    assert list(cell[key])==list(expected[key]) if key=='shape_mkn' else cell[key]==expected[key]
   if cell['status']!='passed':
    assert cell['capped_parity']==0
    continue
   pooled={'native':[],'pytorch':[]}
   assert [e['order'] for e in cell['orders']]==report['policy']['orders']
   for entry in cell['orders']:
    for key,sample in entry['samples'].items():
     calls,blocks=sample['calls_us'],sample['samples_us'];assert len(calls)==sample['checked_calls']==155 and len(blocks)==31
     for i,block in enumerate(blocks):assert math.isclose(sum(calls[5*i:5*i+5])/5,block,rel_tol=1e-12)
     assert statistics.median(blocks)==sample['median_us'];pooled[key].extend(blocks);count+=len(calls)
   for key,values in pooled.items():assert statistics.median(values)==cell['timings'][key]['median_us']
   assert math.isclose(cell['capped_parity'],min(1.,statistics.median(pooled['pytorch'])/statistics.median(pooled['native'])),rel_tol=1e-12)
  assert math.isclose(aggregate(report['cases']),report['capped_geometric_parity'],rel_tol=1e-12)
  receipt=json.loads(path.with_suffix('.receipt.json').read_text())
  assert receipt['commit']==COMMITS[side] and not receipt['status_before'] and not receipt['status_after']
  assert sha256(path.with_suffix('.log'))==receipt['log_sha256']
  reports.append(report)
 baseline,candidate=reports
 for key in ['policy','python','pytorch','pytorch_cuda','harness_sha256','nvcc','rustc']:
  assert baseline[key]==candidate[key],key
 for key in ['triton_version','ptxas_sha256','ptxas_version','pytorch_build']:
  assert baseline['reference_triton_compiler'][key]==candidate['reference_triton_compiler'][key],key
 rows=[]
 for b,a in zip(baseline['cases'],candidate['cases']):
  assert b.get('input_hashes')==a.get('input_hashes')
  row={k:a[k] for k in ['shape_mkn','program','offset','status','capped_parity']}
  row.update(baseline_status=b['status'],baseline_capped_parity=b['capped_parity'],first_call_ms={side:[e.get('first_call_ms') for e in c['orders']] for side,c in [('baseline',b),('candidate',a)]})
  if b['status']==a['status']=='passed':
   row['orders']=[dict(order=x['order'],baseline_us={k:v['median_us'] for k,v in x['samples'].items()},candidate_us={k:v['median_us'] for k,v in y['samples'].items()}) for x,y in zip(b['orders'],a['orders'])]
   row.update(baseline_native_us=b['timings']['native']['median_us'],candidate_native_us=a['timings']['native']['median_us'],baseline_reference_us=b['timings']['pytorch']['median_us'],candidate_reference_us=a['timings']['pytorch']['median_us'],native_speedup=b['timings']['native']['median_us']/a['timings']['native']['median_us'],reference_drift=a['timings']['pytorch']['median_us']/b['timings']['pytorch']['median_us'])
  rows.append(row)
 groups={}
 for kind in ['all','composed','matmul']:
  selected=rows if kind=='all' else [r for r in rows if r['program']==kind]
  groups[kind]={key:geom([r.get(key,0) for r in selected]) for key in ['baseline_native_us','candidate_native_us','native_speedup','reference_drift']}
 pairs.append(dict(name=name,seed=seed,baseline_parity=baseline['capped_geometric_parity'],candidate_parity=candidate['capped_geometric_parity'],groups=groups,cells=rows))
result=dict(measurement_kind='new complete paired clean-code composite capture',commits=COMMITS,raw_calls_verified=count,pairs=pairs)
(OUT/'comparison.json').write_text(json.dumps(result,indent=2)+'\n')
for pair in pairs:print(pair['name'],pair['groups'])
print('Raw calls verified:',count)
