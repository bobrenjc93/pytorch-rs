index={}
def copy(source,relative):
    dest=DEST/relative;dest.parent.mkdir(parents=True,exist_ok=True)
    shutil.copyfile(source,dest)
    digest=hashlib.sha256(source.read_bytes()).hexdigest()
    assert hashlib.sha256(dest.read_bytes()).hexdigest()==digest
    index[relative]={'source':str(source),'sha256':digest}
for path in sorted(OUT.iterdir()):
    if path.is_file() and path.suffix in ('.json','.log','.txt'):
        copy(path,path.name)

    copy(OLD/'baseline-build'/name,'baseline/build/'+name)
for name in ('profile_graph.py','run.py'):
    copy(OLD/name,'baseline/command-sources/'+name+'.txt')
for name in ('held-outs.json','baseline-profile.log','baseline-profile.receipt.json','baseline-profile-retry.log','baseline-profile-retry.receipt.json','matmul.prof','composed.prof'):

drift=[cell['reference_latency_ratio'] for cell in allcells]
