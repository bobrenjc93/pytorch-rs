def git(*args): return subprocess.check_output(['git',*args])
def source_manifest(commit, *paths):
    archive=git('archive','--format=tar',commit,*paths)
    with tarfile.open(fileobj=io.BytesIO(archive)) as tar:
        return {m.name:hashlib.sha256(tar.extractfile(m).read()).hexdigest() for m in tar.getmembers() if m.isfile()}
def code_identity(commit):
    manifest=source_manifest(commit,'src','python','Cargo.toml','Cargo.lock','pyproject.toml','rust-toolchain.toml')
    return hashlib.sha256(json.dumps(manifest,sort_keys=True).encode()).hexdigest()
def native_build(path,commit):
    record=json.loads(path.read_text())
    assert record['commit']==commit and record['clean_checkout'] and not record['git_status_before_build']
    assert record['source_sha256']==code_identity(commit)
    assert record['production_diff_sha256']==hashlib.sha256(b'').hexdigest()
    assert pathlib.Path(record['interpreter_path']).resolve().is_relative_to(ROOT)
    assert sha256(record['interpreter_path'])==record['interpreter_sha256']
    assert sha256(record['rustc_path'])==record['rustc_sha256']
    wheel=pathlib.Path(record['wheel_path']).resolve()
    assert wheel.is_relative_to(ROOT) and sha256(wheel)==record['wheel_sha256']
    with zipfile.ZipFile(wheel) as z:
        native=[n for n in z.namelist() if n.endswith('.so')];assert len(native)==1
        assert hashlib.sha256(z.read(native[0])).hexdigest()==record['extension_sha256']
        sources=source_manifest(commit,'python')
        for path,digest in sources.items():
            if path.endswith('.py'):
                assert hashlib.sha256(z.read(path.removeprefix('python/'))).hexdigest()==digest
    for cmd in record['commands']:
        assert cmd['exit_status']==0 and sha256(cmd['log'])==cmd['log_sha256']
    return record

def inspect_builds(value):
    if isinstance(value,dict):
        if value.get('source_path') and value.get('library_path') and value.get('source_checksum'):
            source=pathlib.Path(value['source_path']).resolve();library=pathlib.Path(value['library_path']).resolve()
            assert source.is_relative_to(ROOT) and library.is_relative_to(ROOT)
            assert hashlib.blake2b(source.read_bytes(),digest_size=16).hexdigest()==value['source_checksum']
            scoring_artifacts[str(library)]=dict(sha256=sha256(library),source_path=str(source),source_sha256=sha256(source),reused=value.get('reused'))
        for item in value.values():inspect_builds(item)
    elif isinstance(value,list):
        for item in value:inspect_builds(item)
