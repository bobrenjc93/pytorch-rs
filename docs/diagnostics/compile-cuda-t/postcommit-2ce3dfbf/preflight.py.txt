import ctypes, importlib.metadata as md, pathlib, subprocess, sys
import numpy, torch, torch_rs
sys.path.insert(0, str(pathlib.Path.cwd()))
from tests.test_cuda_contiguous import runtime
assert pathlib.Path(sys.prefix).resolve() == pathlib.Path('.venv').resolve()
assert torch.__version__.split('+', 1)[0] == '2.13.0'
print('Python', sys.version, sys.executable)
for module in (numpy, torch, torch_rs, torch_rs._C):
    print(module.__name__, getattr(module, '__version__', ''), module.__file__)
print('torch-rs', md.version('torch-rs'))
print('reference CUDA', torch.version.cuda, torch.cuda.device_count(), torch.cuda.get_device_name(0))
version=ctypes.c_int()
assert runtime().cudaRuntimeGetVersion(ctypes.byref(version)) == 0
print('native CUDA runtime', version.value, runtime()._name)
for command in (['rustc', '-Vv'], ['cargo', '-V'], ['nvcc', '--version'], ['nvidia-smi', '--query-gpu=index,uuid,name,driver_version,compute_cap,utilization.gpu,memory.used,memory.total', '--format=csv']):
    print(subprocess.check_output(command, text=True))
print('nvcc unused: embedded PTX is compiled by the CUDA driver')
