import pathlib, sys, ctypes
sys.path.insert(0, str(pathlib.Path.cwd()))
import torch, torch_rs as n
assert pathlib.Path(sys.prefix).resolve() == pathlib.Path('.venv').resolve()
print('python', sys.executable, sys.version)
print('reference', torch.__version__, torch.version.cuda, torch.__file__)
print('native', n.__file__, n._C.__file__)
print('GPU', torch.cuda.get_device_name(0))
from tests.test_cuda_contiguous import runtime
version = ctypes.c_int()
assert runtime().cudaRuntimeGetVersion(ctypes.byref(version)) == 0
print('native runtime', version.value)
def f(x):
    return x.t()
a = n.tensor([[1., 2., 3.], [4., 5., 6.]]).to('cuda:0')
b = torch.tensor([[1., 2., 3.], [4., 5., 6.]], device='cuda:0')
expected = torch.compile(f, backend='eager', fullgraph=True)(b)
assert a.t().cpu().tolist() == expected.cpu().tolist()
print('native eager and reference compiled t agree', expected.shape, expected.stride())
try:
    n.compile(f, backend='eager', fullgraph=True)(a)
except NotImplementedError as error:
    print('EXPECTED BASELINE GAP:', type(error).__name__, str(error))
else:
    raise AssertionError('baseline unexpectedly captures t')
