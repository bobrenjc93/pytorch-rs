import datetime, hashlib, json, subprocess, sys
from pathlib import Path

record={'verified_at':datetime.datetime.now(datetime.timezone.utc).isoformat(),

        'preceding_measured_commit':build['commit'],'preceding_build_record_sha256':sha(old/'release/build-record.json'),
        'preceding_environment_script_exists':(root/'target/reshape-postcommit-73b74ec0/env.sh').exists(),
        'checks':checks,'verification_passed':all(c['matches'] for c in checks.values()),
        'initial_verification_failure':'Sourcing the preceding env.sh failed because it was missing; the following native extension hash assertion failed.',
