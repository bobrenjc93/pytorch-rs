import hashlib, json
from pathlib import Path
import torch_rs
root=Path.cwd()
installed=Path(torch_rs.__file__).parent
checked={}
for path in (root/'python/torch_rs').rglob('*.py'):
    rel=path.relative_to(root/'python/torch_rs')
    other=installed/rel
    assert path.read_bytes()==other.read_bytes(), rel
    checked[str(rel)]=hashlib.sha256(path.read_bytes()).hexdigest()
print(json.dumps({'installed_python_files_verified':len(checked),'files':checked},indent=2))
