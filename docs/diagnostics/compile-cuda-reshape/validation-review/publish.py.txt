"""Preserve review-revision measurements with the established capture helpers."""
import difflib, gzip, hashlib, json, re, shutil, subprocess, sys
from pathlib import Path
ROOT=Path.cwd()
WORK=ROOT/'target/reshape-validation-review'
OUT=ROOT/'docs/diagnostics/compile-cuda-reshape/validation-review'
sys.path.insert(0,str(ROOT/'scripts'))
from evaluate_cuda_math import source_provenance, sha256

def dump(path,value): path.write_text(json.dumps(value,indent=2,sort_keys=True)+'\n')
source=source_provenance()
assert not OUT.exists()
for prior in (OUT.parent,OUT.parent/'postcommit-1ad5fd62',OUT.parent/'binding-review',OUT.parent/'postcommit-7533cbbc'):
    for name,digest in json.loads((prior/'inventory.json').read_text()).items():
        assert sha256(prior/name)==digest,name
build=json.loads((WORK/'verified-release/build-record.json').read_text())
assert all(build[k]==v for k,v in source.items())
assert not build['clean_checkout'] and build['source_unchanged_during_build']
for command in build['commands']:
    assert command['exit_status']==0 and sha256(command['log'])==command['log_sha256']
for field in ('extension','wheel','interpreter'):
    assert Path(build[field+'_path']).resolve().is_relative_to(ROOT)
    assert sha256(build[field+'_path'])==build[field+'_sha256']
final_names=('output-errors','imports','python-imports','preflight','validation-after','reshape-final',
             'compiler-regressions','two-device','cuda-packing','hidden','cpu-layout','docs')
final_checks={name+'-verified' for name in final_names}
expected={name:0 for name in final_checks}
expected.update({'validation-before':1,'imports':0,'python-imports':0,'preflight':0,
                 'validation-after':0,'reshape-final':0,'compiler-regressions':1})
receipts={}
for path in (WORK/'evidence').glob('*.receipt.json'):
    name=path.name.removesuffix('.receipt.json');r=json.loads(path.read_text())
    assert r['exit_status']==expected[name],name
    assert r['source_before']==r['source_after'] and r['inputs_unchanged'],name
    assert r['git_status_before']==r['git_status_after'],name
    assert r['command_files_before']==r['command_files_after'],name
    assert all(sha256(p)==digest for p,digest in r['command_files_after'].items()),name
    assert r['capture_script_sha256']==sha256(WORK/'capture.py'),name
    assert r['native_before']==r['native_after'],name
    assert r['log_sha256']==sha256(WORK/'evidence'/f'{name}.log'),name
    if name in final_checks:
        assert r['source_before']==source,name
        assert set(r['native_after'].values())=={build['extension_sha256']},name
    receipts[name]=r
assert set(receipts)==set(expected)
intermediate=json.loads((WORK/'final-release/build-record.json').read_text())
assert all(intermediate[k]==v for k,v in receipts['compiler-regressions']['source_before'].items())
assert set(receipts['compiler-regressions']['native_before'].values())=={intermediate['extension_sha256']}
for command in intermediate['commands']:
    assert command['exit_status']==0 and sha256(command['log'])==command['log_sha256']
import torch_rs
installed=Path(torch_rs.__file__).parent
for path in (ROOT/'python/torch_rs').rglob('*.py'):
    assert sha256(path)==sha256(installed/path.relative_to(ROOT/'python/torch_rs'))
assert sha256(torch_rs._C.__file__)==build['extension_sha256']
manifests={sha256(p):json.loads(p.read_text()) for p in (WORK/'evidence').glob('inputs-*.json')}
final_hash=receipts['reshape-final-verified']['measured_inputs_sha256'];manifest=manifests[final_hash]
assert all(sha256(ROOT/p)==digest for p,digest in manifest.items())
results={}
for name in ('output-errors','validation-after','reshape-final','compiler-regressions','two-device','cuda-packing','hidden','cpu-layout'):
    name+='-verified'
    text=(WORK/'evidence'/f'{name}.log').read_text()
    count=int(re.search(r'Ran (\d+) tests? in ',text).group(1))
    match=re.search(r'OK \(skipped=(\d+)\)',text)
    skipped=int(match.group(1)) if match else 0
    assert re.search(r'\nOK(?: \(skipped=\d+\))?\s*$',text),name
    results[name]={'passed':count-skipped,'skipped':skipped}
# The reproduction predates two additional unsupported-unused-tuple assertions.
test_path='tests/test_compile_cuda_reshape.py'
current=(ROOT/test_path).read_text()
first=''.join(line for line in current.splitlines(keepends=True) if not line.lstrip().startswith("'unused = (1, x)"))
first_hash=manifests[receipts['validation-before']['measured_inputs_sha256']][test_path]
assert hashlib.sha256(first.encode()).hexdigest()==first_hash
OUT.mkdir()
(OUT/'release').mkdir()
for name in ('build-record.json','commands.json','build.log','install.log'):
    shutil.copyfile(WORK/'verified-release'/name,OUT/'release'/name)
(OUT/'intermediate-release').mkdir()
for name in ('build-record.json','commands.json','build.log','install.log'):
    shutil.copyfile(WORK/'final-release'/name,OUT/'intermediate-release'/name)
for name in receipts:
    for suffix in ('.log','.receipt.json'):
        if name=='validation-before' and suffix=='.log':
            raw=(WORK/'evidence'/(name+suffix)).read_bytes()
            compressed=gzip.compress(raw,mtime=0)
            assert gzip.decompress(compressed)==raw
            (OUT/(name+suffix+'.gz')).write_bytes(compressed)
        else:
            shutil.copyfile(WORK/'evidence'/(name+suffix),OUT/(name+suffix))
dump(OUT/'measured-inputs.json',manifest)
index={}
for digest,measured in manifests.items():
    if digest==final_hash:
        index[digest]={'file':'measured-inputs.json'};continue
    delta={p:h for p,h in measured.items() if manifest.get(p)!=h}
    removed=sorted(set(manifest)-set(measured))
    name='inputs-delta-'+digest[:12]+'.json'
    dump(OUT/name,{'replace':delta,'remove':removed})
    recovered={p:h for p,h in manifest.items() if p not in removed}|delta
    assert hashlib.sha256((json.dumps(recovered,indent=2,sort_keys=True)+'\n').encode()).hexdigest()==digest
    index[digest]={'base':'measured-inputs.json','delta':name}
dump(OUT/'manifest-index.json',index)
(OUT/'reproduction-tests.patch').write_text(''.join(difflib.unified_diff(current.splitlines(keepends=True),first.splitlines(keepends=True),fromfile='a/'+test_path,tofile='b/'+test_path)))
(OUT/'revision.patch').write_bytes(subprocess.check_output(['git','diff','HEAD','--','python/torch_rs/_compile_bytecode.py','python/torch_rs/_compile_trace.py',test_path]))
failed_path='python/torch_rs/_compile_bytecode.py'
saved=WORK/'before-output-diagnostic.py.txt'
assert sha256(saved)==manifests[receipts['compiler-regressions']['measured_inputs_sha256']][failed_path]
(OUT/'output-diagnostic.patch').write_text(''.join(difflib.unified_diff(
    (ROOT/failed_path).read_text().splitlines(keepends=True),saved.read_text().splitlines(keepends=True),
    fromfile='a/'+failed_path,tofile='b/'+failed_path)))
for name in ('capture.py','validate.py','validate-verified.py','publish.py'):

    shutil.copyfile(WORK/name,OUT/(name+'.txt'))
for name in ('env.sh','preflight.py','check_install.py','docs_example.py'):
    shutil.copyfile(ROOT/'target/reshape-postcommit-7533cbbc'/name,OUT/(name+'.txt'))
for name in ('validation.log','validation-verified.log'):
    shutil.copyfile(WORK/name,OUT/name)
rows=[]
for name,result in results.items():
    rows.append(f"| [{name}]({name}.log) | {result['passed']} passed; {result['skipped']} skipped |")
(OUT/'README.md').write_text(f"""# Reshape partial-argument validation revision

These are **development measurements**, based on `{source['commit']}` plus
[the recorded Python/test revision](revision.patch), with source hash
`{source['source_sha256']}`. A fresh final release wheel was built with the
existing [build helper](../../../../scripts/capture_depth_concat_build.py).
Installed Python files and native imports were verified before the final tests.
The canonical worktree-local CPython 3.12.14 environment, locked dependencies,
CUDA caches and local Cargo registry were reused; the release build target was
empty. See the [build record](release/build-record.json), [preflight](preflight-verified.log),
command receipts and [runner](validate-verified.py.txt). All writes stayed in this worktree.

The differential regression covers known-invalid bool/float/overflow dimensions
alongside Tensor arguments, mixed literal/local tuples with invalid binding,
all four policies, early/late nodes, repeated rejected calls and empty caches.
It blocks native execution and Python-body replay. Nonconstant shapes, mixed
output pytrees, tuple helper arguments/arithmetic and unused mixed tuples stay
unsupported. Unknown values are opaque during validation, with no conversion.

| Check | Result |
| --- | --- |
"""+'\n'.join(rows)+"""

Installed imports, source/installed Python byte equality and the guide example
also passed. This revision changes Python validation only; native planners and
kernels are unchanged. Ordinary CUDA checks use physical GPU 0, an H100 with
driver 580.82.07 and native/reference CUDA 13.0 (PyTorch 2.13.0+cu130); only
context/restoration checks expose 0,1. Build metadata records Rust 1.92.0;
nvcc 12.6.85 was installed but unused. Receipts include physical UUID, utilization
and memory snapshots, which do not reserve hardware.

The [pre-fix reproduction log](validation-before.log.gz) preserves all 256 errors
losslessly compressed with gzip. Its decompressed SHA-256 matches the unchanged
[receipt](validation-before.receipt.json); [reproduction-tests.patch](reproduction-tests.patch)
reconstructs that test revision. The first [compiler regression run](compiler-regressions.log)
also preserved two failures: rejection of scalar output leaves lost its established
leaf-specific diagnostic. The implementation restored that diagnostic without
changing the pre-existing assertions. [output-diagnostic.patch](output-diagnostic.patch)
reconstructs the failed implementation, bound to [its build](intermediate-release/build-record.json).
The verified run and table above use a new release build after that correction.
Neither failed run is counted as passing.
One [input manifest](measured-inputs.json) and verified deltas cover all attempts;
[manifest-index](manifest-index.json) maps every receipt hash. The
[publisher](publish.py.txt) checks source/build/import/log hashes before copying.

All prior development failures and clean captures remain unchanged and pinned
to their original code, including [the preceding clean capture](../postcommit-7533cbbc/README.md).
They do not validate this revision. **A fresh clean-commit capture remains required
after Burner commits this fix**; this development bundle does not satisfy that step.
Scoring corpora, evaluators, workloads, managed progress and Burner contracts are
unchanged. Independent review and merge gates remain required.
""")
dump(OUT/'audit.json',{'source':source,'measurement_kind':'precommit-diagnostic','checks':expected,
 'results':results,'installed_python_matches_source':True,'installed_native_matches_build':True,
 'extension_sha256':build['extension_sha256'],'previous_inventories_unchanged':True,
 'clean_commit_capture':'required after Burner commits this revision',
 'manifest_count':len(manifests),'full_manifests_published':1})
dump(OUT/'inventory.json',{str(p.relative_to(OUT)):sha256(p) for p in OUT.rglob('*') if p.is_file() and p.name!='inventory.json'})
print(json.dumps(results,indent=2))
print('Published',OUT)
