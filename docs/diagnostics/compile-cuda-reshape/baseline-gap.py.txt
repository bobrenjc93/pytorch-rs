import sys
from pathlib import Path
sys.path.insert(0, str(Path.cwd()))
import torch, torch_rs as n
from tests.test_compile_cuda_mul_scalar import make_program
from tests.test_cuda_contiguous import metadata
print('python',sys.version,sys.executable)
print('imports',n.__file__, n._C.__file__,torch.__file__)
print('reference',torch.__version__,torch.version.cuda,torch.cuda.get_device_name(0))
f=make_program('def program(x):\n    return x.reshape(-1)\n')
for transpose in (False,True):
 a=n.tensor([[1.,2.,3.],[4.,5.,6.]]).to('cuda:0')
 b=torch.tensor([[1.,2.,3.],[4.,5.,6.]],device='cuda:0')
 if transpose: a,b=a.t(),b.t()
 ref=torch.compile(f,backend='eager',fullgraph=True)(b)
 eager=f(a)
 print('input',metadata(a),'reference compiled',ref.cpu().tolist(),'native eager',eager.cpu().tolist(), 'alias', eager.data_ptr()==a.data_ptr())
 assert eager.cpu().tolist()==ref.cpu().tolist()
 try: n.compile(f,backend='eager',fullgraph=True)(a)
 except NotImplementedError as e: print('confirmed baseline gap',type(e).__name__,str(e))
 else: raise AssertionError('expected missing reshape capture')
