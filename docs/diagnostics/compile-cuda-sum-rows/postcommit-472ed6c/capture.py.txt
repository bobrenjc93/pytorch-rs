def git(*args):
    return subprocess.check_output(['git', *args], cwd=ROOT).decode().strip()


def write(name, value):
    (OUT / name).write_text(json.dumps(value, indent=2, sort_keys=True) + '\n')


SOURCE_PATHS = git('ls-files', '--', 'src', 'python', 'tests', 'scripts', '.github',
                  'Cargo.toml', 'Cargo.lock', 'pyproject.toml', 'rust-toolchain.toml',
                  'docs/hardware-heterogeneity-matrix-v1.json').splitlines()
SOURCE_FILES = {p: sha256(ROOT / p) for p in SOURCE_PATHS}
write('source-files.json', SOURCE_FILES)


def snapshot():
    result = {'commit': git('rev-parse', 'HEAD'), 'tree': git('rev-parse', 'HEAD^{tree}'),
              'status': git('status', '--porcelain', '--untracked-files=all'),
              'production': source_provenance(),
              'source_files_sha256': hashlib.sha256(json.dumps(
                  {p: sha256(ROOT / p) for p in SOURCE_PATHS}, sort_keys=True).encode()).hexdigest()}
    assert result['commit'] == COMMIT and result['status'] == '', result
    return result


BASELINE = snapshot()
write('clean-snapshot.json', BASELINE)
ENV_KEYS = ('CARGO_HOME', 'CARGO_TARGET_DIR', 'PYO3_PYTHON', 'RUSTC', 'RUSTDOC',
            'TMPDIR', 'XDG_CACHE_HOME', 'CUDA_CACHE_PATH', 'TORCHINDUCTOR_CACHE_DIR',

            'CUDA_VISIBLE_DEVICES', 'PYTHONPATH')
write('environment.json', {k: os.environ.get(k) for k in ENV_KEYS})


def run(name, command, overrides=None, gpu=False, timeout=600):
    env = {**os.environ, **(overrides or {})}
    env = {k: v for k, v in env.items() if v is not None}
    before = snapshot()
    assert before == BASELINE
    inventory = ['nvidia-smi', '--query-gpu=index,uuid,name,memory.total,memory.used,utilization.gpu,driver_version,compute_cap', '--format=csv']
    if gpu:
        (OUT / f'{name}.gpu-before.csv').write_bytes(subprocess.check_output(inventory))
    started = time.time()
    error = None
    with (OUT / f'{name}.stdout.log').open('w') as stdout, (OUT / f'{name}.stderr.log').open('w') as stderr:
        try:
            completed = subprocess.run(command, cwd=ROOT, env=env, stdout=stdout, stderr=stderr, timeout=timeout)
            code = completed.returncode
        except subprocess.TimeoutExpired as exc:
            error, code = str(exc), 124
    if gpu:
        (OUT / f'{name}.gpu-after.csv').write_bytes(subprocess.check_output(inventory))
    after = snapshot()
    record = {'command': command, 'cwd': str(ROOT), 'environment': {k: env.get(k) for k in ENV_KEYS},
              'started_unix': started, 'finished_unix': time.time(), 'exit_code': code, 'error': error,
              'before': before, 'after': after, 'source_unchanged': before == after,
              'stdout_sha256': sha256(OUT / f'{name}.stdout.log'),
              'stderr_sha256': sha256(OUT / f'{name}.stderr.log')}
    extension = ROOT / 'python/torch_rs/torch_rs.abi3.so'
    if extension.exists():
        record['extension_sha256'] = sha256(extension)
    write(f'{name}.receipt.json', record)
    print(name, code, flush=True)
    assert before == after
    if code:
        raise RuntimeError(f'{name} failed with exit {code}; retained raw outputs')
    return record
