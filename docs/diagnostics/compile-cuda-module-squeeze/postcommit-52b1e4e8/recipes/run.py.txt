"""Timestamp and clean-commit envelope around the committed receipt tool."""
from datetime import datetime, timezone
import hashlib, json, pathlib, subprocess, sys
root=pathlib.Path.cwd()
commit='52b1e4e8ed45c6f78d66ff790fe926bb08607c28'
out=root/'target/postcommit-52b1e4e8'
label,*command=sys.argv[1:]
def git(*args): return subprocess.check_output(['git',*args],text=True)
def stamp(): return datetime.now(timezone.utc).isoformat()
assert git('rev-parse','HEAD').strip()==commit
before=git('status','--porcelain')
assert not before, before
name='postcommit-52b1e4e8-'+label
path=out/(label+'.json')
assert not path.exists()
started=stamp()
result=subprocess.run([sys.executable,'docs/diagnostics/compile-cuda-module-squeeze/recipes/record.py.txt',name,*command])
finished=stamp()
after=git('status','--porcelain')
original=root/'target/module-squeeze-evidence'/(name+'.json')
record=json.loads(original.read_text())
record.update(measured_code_commit=commit,started_at=started,finished_at=finished,
              origin_status_before=before,origin_status_after=after,
              original_receipt_sha256=hashlib.sha256(original.read_bytes()).hexdigest())
path.write_text(json.dumps(record,indent=2)+'\n')
assert not after,after
sys.exit(result.returncode)
