"""Verify the published clean capture and restrict the delivery diff to evidence."""
from datetime import datetime, timezone
import gzip, hashlib, json, pathlib, subprocess
root=pathlib.Path.cwd()
commit='52b1e4e8ed45c6f78d66ff790fe926bb08607c28'
relative='docs/diagnostics/compile-cuda-module-squeeze/postcommit-52b1e4e8/'
out=root/relative
def sha(raw): return hashlib.sha256(raw).hexdigest()
def git(*args): return subprocess.check_output(['git',*args],text=True)
assert git('rev-parse','HEAD').strip()==commit
changed=git('diff','--name-only').splitlines()
assert changed==['docs/compile-cuda-squeeze.md'],changed
assert all(p.startswith(relative) for p in git('ls-files','--others','--exclude-standard').splitlines())
committed=json.loads((root/'target/postcommit-52b1e4e8/committed-files.json').read_text())
for p,digest in committed.items():
    if p not in changed: assert sha((root/p).read_bytes())==digest,p
commands=json.loads((out/'commands.json').read_text())
capture=json.loads((out/'capture.json').read_text())
build_raw=gzip.decompress((out/'build-record.json.gz').read_bytes())
build=json.loads(build_raw)
assert sha(build_raw)==capture['build_record_sha256']
assert build['measured_code_commit']==commit and build['origin_status']==''
assert build['source_matches_commit'] and build['origin_patch_sha256']==sha(b'')
for key in ('source_root','wheel','installed_package','installed_native','build_target'):
    assert pathlib.Path(build[key]).resolve().is_relative_to(root),key
assert sha(pathlib.Path(build['wheel']).read_bytes())==build['wheel_sha256']
assert sha(pathlib.Path(build['installed_native']).read_bytes())==build['installed_native_sha256']
assert sha(json.dumps(build['source_files'],sort_keys=True).encode())==build['source_manifest_sha256']
assert all(sha((pathlib.Path(build['source_root'])/p).read_bytes())==v for p,v in build['source_files'].items())
extra={'measured_code_commit','started_at','finished_at','origin_status_before','origin_status_after',
       'original_receipt_sha256','log','summary'}
for name,record in commands.items():
    assert record['measured_code_commit']==commit and record['base_commit']==commit
    assert record['origin_status_before']==record['origin_status_after']==''
    assert pathlib.Path(record['cwd']).resolve()==root
    assert record['exit_code']==0 and record['production_unchanged']
    assert sha(gzip.decompress((out/record['log']).read_bytes()))==record['log_sha256']
    original={k:v for k,v in record.items() if k not in extra}
    assert sha((json.dumps(original,indent=2)+'\n').encode())==record['original_receipt_sha256'],name
    for key in ('UV_CACHE_DIR','UV_PYTHON_INSTALL_DIR','UV_PYTHON_BIN_DIR','UV_PROJECT_ENVIRONMENT',
                'TMPDIR','XDG_CACHE_HOME','CUDA_CACHE_PATH','TORCHINDUCTOR_CACHE_DIR','TRITON_CACHE_DIR',
                'PYTHONPYCACHEPREFIX','VIRTUAL_ENV','PYO3_PYTHON','TORCH_RS_CUDART','CARGO_HOME','CARGO_TARGET_DIR'):
        assert pathlib.Path(record['environment'][key]).resolve().is_relative_to(root),(name,key)
    for p,v in record['native'].items():
        assert (root/p).resolve().is_relative_to(root/'.venv')
        if name != 'release-build': assert sha((root/p).read_bytes())==v
for kind,ref in capture['manifest_references'].items():
    index=json.loads((out/ref['index']).read_text())
    files=json.loads(gzip.decompress((out/ref['base']).read_bytes()))
    if ref['sha256']!=index['base_sha256']:
        delta=index['variants'][ref['sha256']]
        for p in delta['removed']: del files[p]
        files.update(delta['changed'])
    assert sha(json.dumps(files,sort_keys=True).encode())==ref['sha256']
    assert all(sha((root/p).read_bytes())==v and build['source_files'][p]==v for p,v in files.items())
    assert all(r[kind+'_manifest']==ref['sha256'] for r in commands.values())
environment=json.loads(gzip.decompress((out/'logs/environment.log.gz').read_bytes()))
assert environment['torch']=='2.13.0+cu130' and environment['versions']['cudaRuntimeGetVersion']==13000
assert environment['gpu']=='NVIDIA H100' and '12.6.85' in environment['nvcc']
assert environment['installed_python_matches_source'] and environment['native_sha256']==build['installed_native_sha256']
dependencies=json.loads(gzip.decompress((out/'logs/dependency-paths.log.gz').read_bytes()))
for entry in dependencies.values():
    path=pathlib.Path(entry['file'])
    assert path.resolve().is_relative_to(root/'.venv')
    assert sha(path.read_bytes())==entry['sha256']
assert dependencies['reference']['version']==environment['torch']
assert dependencies['reference']['cuda']=='13.0'
selected={p for p in committed if p.startswith('tests/test_compile') and p.endswith('.py')}|{'tests/test_top_level_compile.py'}
assert len(selected)==61
sweep=gzip.decompress((out/'logs/compiler-complete.log.gz').read_bytes()).decode()
assert all(('tests.'+pathlib.Path(p).stem) in sweep for p in selected)
for name in ('provenance-before','provenance-after'):
    p=json.loads(gzip.decompress((out/('logs/'+name+'.log.gz')).read_bytes()))
    assert p['measured_code_commit']==commit and p['origin_status']==''
    assert p['committed_files_sha256']==sha(json.dumps(committed,sort_keys=True).encode())
    assert p['interpreter_inventory_sha256']=='519ae888f6f47836a48399b7eff35b28582673e893a08261bd5e2758f9a4f265'
print(json.dumps(dict(verified_at=datetime.now(timezone.utc).isoformat(),measured_code_commit=commit,
    passed=True,clean_measurement_commands=len(commands),compiler_modules=len(selected),
    source_build_native_input_bindings_verified=True,current_paths_inside_worktree=True,
    historical_records_unchanged=True,implementation_dependencies_tests_and_harnesses_unchanged=True,
    publication_diff=changed,publication_untracked_prefix=relative,
    independent_review_and_merge_gates_still_required=True),indent=2))
