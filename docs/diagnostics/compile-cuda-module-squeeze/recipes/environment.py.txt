import ctypes, hashlib, json, os, pathlib, subprocess, sys, zipfile
import torch, torch_rs
root=pathlib.Path.cwd()
lib=ctypes.CDLL(os.environ['TORCH_RS_CUDART'])
versions={}
for name in ('cudaRuntimeGetVersion','cudaDriverGetVersion'):
    v=ctypes.c_int()
    assert getattr(lib,name)(ctypes.byref(v)) == 0
    versions[name]=v.value
p=pathlib.Path(torch_rs._C.__file__)
assert p.resolve().is_relative_to(root/'.venv')
assert pathlib.Path(sys.prefix).resolve()==root/'.venv'
installed=pathlib.Path(torch_rs.__file__).parent
for file in (root/'python/torch_rs').rglob('*.py'):
    assert file.read_bytes()==(installed/file.relative_to(root/'python/torch_rs')).read_bytes(),file
print(json.dumps(dict(python=sys.version,executable=sys.executable,base_prefix=sys.base_prefix,
    native=str(p),native_sha256=hashlib.sha256(p.read_bytes()).hexdigest(),
    installed_python_matches_source=True,torch=torch.__version__,torch_cuda=torch.version.cuda,
    cuda_runtime=os.environ['TORCH_RS_CUDART'],versions=versions,gpu=torch.cuda.get_device_name(0),
    rust=subprocess.check_output(['rustc','--version'],text=True),nvcc=subprocess.check_output(['nvcc','--version'],text=True),
    build_profile='release, thin LTO, codegen-units=1, extension-module, locked', kernel_compiler='driver JIT of embedded PTX; nvcc not invoked by native build',
    resources=['gpu','cpu-heavy'],reservation=False),indent=2))
