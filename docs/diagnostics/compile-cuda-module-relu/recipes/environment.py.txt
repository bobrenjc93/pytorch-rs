import ctypes
import json
import os
from pathlib import Path
import subprocess
import sys
import torch
import torch_rs
lib = ctypes.CDLL(os.environ['TORCH_RS_CUDART'])
version = ctypes.c_int()
assert lib.cudaRuntimeGetVersion(ctypes.byref(version)) == 0
print(json.dumps(dict(python=sys.version, executable=sys.executable, prefix=sys.prefix,
    torch=torch.__version__, reference_cuda=torch.version.cuda, native=torch_rs._C.__file__,
    runtime=os.environ['TORCH_RS_CUDART'], runtime_version=version.value,
    nvcc=subprocess.check_output(['nvcc', '--version'], text=True),
    rustc=subprocess.check_output(['rustc', '-Vv'], text=True),
    resources=['gpu', 'cpu-heavy'], nvcc_used_for_native_build=False,
    native_kernel_compiler='driver JIT of embedded PTX',
    gpu=subprocess.check_output(['nvidia-smi', '--query-gpu=index,uuid,name,driver_version,compute_cap,memory.total,memory.used,utilization.gpu', '--format=csv'],text=True),
    snapshot_is_not_reservation=True),indent=2))
