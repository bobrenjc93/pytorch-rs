"""Audit development capture integrity; never evaluates or generates progress."""
import gzip
import hashlib
import json
from pathlib import Path
import subprocess
import zipfile

root=Path.cwd().resolve()
folder=root/'docs/diagnostics/compile-cuda-module-relu'
sha=lambda raw:hashlib.sha256(raw).hexdigest()
def read(name): return json.loads((folder/name).read_text())
states=read('input-manifests.json')
for key, delta in states['variants'].items():
    data={k:v for k,v in states['base'].items() if k not in delta['removed']}
    data.update(delta['changed'])
    assert sha(json.dumps(data,sort_keys=True).encode()) == key
commands=read('commands.json')
for label, record in commands.items():
    raw=(folder/record['published_log']).read_bytes()
    if record['published_log'].endswith('.gz'): raw=gzip.decompress(raw)
    assert sha(raw)==record['log_sha256'], label
    assert record['source_manifest_sha256'] in states['variants'], label
final=commands['compiler-complete']
assert final['exit_code']==0
key=final['source_manifest_sha256']; delta=states['variants'][key]
inputs={k:v for k,v in states['base'].items() if k not in delta['removed']}; inputs.update(delta['changed'])
assert all(sha((root/path).read_bytes())==digest for path,digest in inputs.items())
builds=read('builds.json')
base=json.loads(gzip.decompress((folder/'build-source-base.json.gz').read_bytes()))
for name, build in builds.items():
    delta=build['source_delta_from_base']
    files={k:v for k,v in base.items() if k not in delta['removed']}; files.update(delta['changed'])
    assert sha(json.dumps(files,sort_keys=True).encode()) == build['source_manifest_sha256']
    for path,digest in build['logs'].items(): assert sha((folder/path).read_bytes())==digest
    wheel=Path(build['wheel'])
    assert wheel.is_relative_to(root)
    assert sha(wheel.read_bytes())==build['wheel_sha256']
    with zipfile.ZipFile(wheel) as z:
        native,=[path for path in z.namelist() if path.endswith('.so')]
        assert sha(z.read(native))==build['installed_native_sha256']
        for path,digest in files.items():
            if path.startswith('python/') and path.endswith('.py'):
                assert sha(z.read(path.removeprefix('python/')))==digest
    if name=='development':
        assert sha(Path(build['installed_native']).read_bytes())==build['installed_native_sha256']
        for path,digest in files.items():
            if path.startswith(('src/','python/')) or path in ('Cargo.toml','Cargo.lock','pyproject.toml','rust-toolchain.toml','uv.lock'):
                assert sha((root/path).read_bytes())==digest, path
                if path.startswith('python/') and path.endswith('.py'):
                    installed=root/'.venv/lib/python3.12/site-packages'/path.removeprefix('python/')
                    assert sha(installed.read_bytes())==digest
assert sha((folder/'recipes/probe-baseline.py.txt').read_bytes())=='06f27f9ae4a909a7173a1fe16c10de7f172bd8ef9408b963d5db979c8c8674b3'
assert sha((folder/'baseline.json').read_bytes())==commands['baseline']['log_sha256']
assert sha((folder/'audited-baseline.json').read_bytes())=='d2e61a78baade64c898d7d87fdb2f2819584886925c580fdf290a58982442478'
assert read('baseline.json')['gap_cases']==32
assert read('baseline.json')['existing_positive_cases']==12
head=subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip()
assert head=='b74c2051b823518cc09d5a29f8237a3de9863516'
allowed={'FEATURES.md','docs/README.md','docs/compile-cuda-add.md','docs/cuda-relu.md','docs/supported-surface.md',
         'python/torch_rs/_compiler_state.py','python/torch_rs/_compile_bytecode.py',
         'tests/test_compile_cuda_module_arithmetic.py','tests/test_compile_cuda_relu.py'}
changed=set(subprocess.check_output(['git','diff','--name-only'],text=True).splitlines())
assert changed <= allowed, changed-allowed
# New paths must be only this feature test and development evidence.
untracked=subprocess.check_output(['git','ls-files','--others','--exclude-standard'],text=True).splitlines()
assert all(p=='tests/test_compile_cuda_module_relu.py' or p.startswith('docs/diagnostics/compile-cuda-module-relu/') for p in untracked)
print(json.dumps(dict(verified=True,command_count=len(commands),
    final_compiler_source_matches=True,release_wheel_and_installed_python_native_match=True,
    documentation_sha256={name:sha((root/name).read_bytes()) for name in ('FEATURES.md','docs/README.md','docs/compile-cuda-add.md','docs/cuda-relu.md','docs/supported-surface.md')},
    previous_evidence_evaluators_and_managed_progress_unchanged=True,
    source_commit=head,source_kind='uncommitted implementation; development capture',
    clean_commit_capture_complete=False,clean_commit_dependency='Burner-owned implementation commit, review and delivery',
    artifact_sha256={str(p.relative_to(folder)):sha(p.read_bytes()) for p in sorted(folder.rglob('*')) if p.is_file() and p.name!='verification.json'}),indent=2))
