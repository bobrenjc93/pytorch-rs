"""Verify clean-commit ReLU evidence and unchanged implementation/history."""
from datetime import datetime, timezone
import gzip
import hashlib
import json
from pathlib import Path
import subprocess
import zipfile

root=Path.cwd().resolve()
prior=root/'docs/diagnostics/compile-cuda-module-relu'
folder=prior/'postcommit-1abe675f'
head='1abe675fc47215afba609f54716353730237c710'
sha=lambda raw:hashlib.sha256(raw).hexdigest()
def read(name): return json.loads((folder/name).read_text())
assert subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip()==head
commands=read('commands.json')
assert len(commands)==16
for name,record in commands.items():
 raw=(folder/record['published_log']).read_bytes()
 if record['published_log'].endswith('.gz'): raw=gzip.decompress(raw)
 assert sha(raw)==record['log_sha256'],name
 assert record['exit_code']==0,name
 assert record['measured_code_commit']==head,name
 assert record['git_status_before']==record['git_status_after']=='',name
 assert record['started_at']<=record['finished_at'],name
 assert Path(record['cwd']).resolve()==root,name
 assert record['recorder_sha256']==sha((folder/'recipes/run.py.txt').read_bytes()),name
 for path in record['native_extensions']:
  assert (root/path).resolve().is_relative_to(root)
inputs=read('inputs.json')
path=folder/inputs['manifest']
assert sha(path.read_bytes())==inputs['manifest_file_sha256']
states=json.loads(path.read_text()); key=inputs['variant_sha256']; delta=states['variants'][key]
values={k:v for k,v in states['base'].items() if k not in delta['removed']}; values.update(delta['changed'])
assert sha(json.dumps(values,sort_keys=True).encode())==key
assert all(sha((root/name).read_bytes())==digest for name,digest in values.items())
assert {r['source_manifest_sha256'] for r in commands.values()}=={key}
assert sha((folder/inputs['input_recipe']).read_bytes())==inputs['input_recipe_sha256']
build=read('build.json')
assert build['measured_code_commit']==head and build['source_matches_commit']
assert build['origin_status']=='' and build['source_kind']=='clean commit export'
raw=gzip.decompress((folder/build['base_source_manifest']).read_bytes())
assert sha(raw)==build['base_source_manifest_sha256']
base=json.loads(raw); delta=build['source_delta_from_base']
files={k:v for k,v in base.items() if k not in delta['removed']}; files.update(delta['changed'])
assert sha(json.dumps(files,sort_keys=True).encode())==build['source_manifest_sha256']
for key in ('source_root','wheel','installed_package','installed_native','build_target'):
 assert Path(build[key]).resolve().is_relative_to(root),(key,build[key])
assert all(sha((Path(build['source_root'])/name).read_bytes())==digest for name,digest in files.items())
assert sha(Path(build['wheel']).read_bytes())==build['wheel_sha256']
assert sha(Path(build['installed_native']).read_bytes())==build['installed_native_sha256']
with zipfile.ZipFile(build['wheel']) as wheel:
 native,=[name for name in wheel.namelist() if name.endswith('.so')]
 assert sha(wheel.read(native))==build['installed_native_sha256']
 for name,digest in files.items():
  if name.startswith('python/') and name.endswith('.py'):
   member=name.removeprefix('python/')
   assert sha(wheel.read(member))==digest
   assert sha((root/'.venv/lib/python3.12/site-packages'/member).read_bytes())==digest
for name,digest in build['logs'].items(): assert sha((folder/name).read_bytes())==digest
assert sha((Path(build['source_root']).parent/'build-record.json').read_bytes())==build['original_record_sha256']
preflight=read('preflight.json')
assert preflight['verified'] and preflight['measured_code_commit']==head
assert preflight['source_provenance']['production_diff_sha256']==sha(b'')
assert sha((folder/'preflight.json').read_bytes())==commands['preflight']['log_sha256']
if (folder/'publication-docs.json').exists():
 publication=read('publication-docs.json')
 assert publication['exit_code']==0 and publication['measured_code_commit']==head
 assert sha((folder/publication['published_log']).read_bytes())==publication['log_sha256']
 for path,digest in publication['documentation_sha256'].items():
  assert sha((root/path).read_bytes())==digest,path
# The original baseline and development files keep all their original bytes.
old=json.loads((prior/'verification.json').read_text())
for name,digest in old['artifact_sha256'].items(): assert sha((prior/name).read_bytes())==digest,name
for name,entry in read('recipes.json')['unchanged_recipes'].items():
 assert sha((folder/entry['path']).read_bytes())==entry['sha256'],name
# Publication may add evidence and an accompanying compiler-guide link only.
changed=set(subprocess.check_output(['git','diff','--name-only'],text=True).splitlines())
assert changed<={'docs/compile-cuda-add.md'},changed
untracked=subprocess.check_output(['git','ls-files','--others','--exclude-standard'],text=True).splitlines()
assert all(p.startswith('docs/diagnostics/compile-cuda-module-relu/postcommit-1abe675f/') for p in untracked)
print(json.dumps(dict(verified=True,checked_at=datetime.now(timezone.utc).isoformat(),
 measured_code_commit=head,clean_commit_capture_complete=True,command_count=len(commands),
 source_and_installed_wheel_verified=True,prior_evidence_unchanged=True,
 implementation_tests_harnesses_locks_and_managed_progress_unchanged=True,
 source_kind='clean committed implementation; later publication is evidence/documentation only',
 artifact_sha256={str(p.relative_to(folder)):sha(p.read_bytes()) for p in sorted(folder.rglob('*')) if p.is_file() and p.name!='verification.json'}),indent=2))
