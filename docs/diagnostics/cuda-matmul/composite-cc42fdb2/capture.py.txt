"""Evidence command orchestration; all workloads use committed repository code."""
from pathlib import Path
import hashlib
import importlib.util
import json
import os
import subprocess
import sys

ROOT=Path.cwd().resolve()
OUT=ROOT/'target/postcommit-cc42fdb2'
COMMIT='cc42fdb265af856d8110404e91ff6cf8dd70a5e7'
spec=importlib.util.spec_from_file_location('capture',ROOT/'docs/diagnostics/composite-cuda-neg/reproduce.py')
capture=importlib.util.module_from_spec(spec)
spec.loader.exec_module(capture)
capture.OUT=OUT

def git(*args):
    return subprocess.check_output(['git',*args],cwd=ROOT,text=True)

def clean():
    assert git('rev-parse','HEAD').strip()==COMMIT
    assert git('status','--porcelain')=='',git('status','--porcelain')

def hashes():
    files=git('ls-files','--','src','python','tests','scripts','Cargo.toml','Cargo.lock','pyproject.toml','uv.lock','rust-toolchain.toml').splitlines()
    return {path:capture.sha256(ROOT/path) for path in files}

clean()
initial=hashes()
assert not Path(os.environ['CARGO_TARGET_DIR']).exists()
state=ROOT/'target/postcommit-cc42fdb2-state'
assert not state.exists()
for key in ('TMPDIR','CUDA_CACHE_PATH','XDG_CACHE_HOME','TRITON_CACHE_DIR','TORCHINDUCTOR_CACHE_DIR','MPLCONFIGDIR'):
    path=Path(os.environ[key]).resolve()
    assert path.is_relative_to(ROOT)
    path.mkdir(parents=True,exist_ok=False)
for key in ('CARGO_HOME','PYO3_PYTHON','TORCH_RS_CUDART','TORCH_RS_CUBLAS'):
    assert Path(os.environ[key]).resolve().is_relative_to(ROOT)
assert Path(sys.executable).resolve().is_relative_to(ROOT)
assert Path(sys.base_prefix).resolve().is_relative_to(ROOT)
changed=git('diff','--name-only','96205cb01e85f6961a4d8ea40aa792b485289d84',COMMIT).splitlines()
capture.write('source-record.json',dict(commit=COMMIT,base='96205cb01e85f6961a4d8ea40aa792b485289d84',
    started_at=capture.stamp(),root=str(ROOT),git_status=git('status','--porcelain'),
    source_provenance=capture.source_provenance(),code_test_harness_hashes=initial,
    candidate_files=changed,historical_source_evidence={p:capture.sha256(ROOT/p) for p in changed if p.startswith('docs/diagnostics/cuda-matmul/')},
    executable=sys.executable,resolved_executable=str(Path(sys.executable).resolve()),base_prefix=sys.base_prefix,
    build_target_initially_absent=True,state_directory_initially_absent=True,
    reused='worktree-local Python packages/interpreter, previously fetched worktree-local Cargo registry; external Rust toolchain and driver inspected/executed read-only',
    environment={key:os.environ.get(key) for key in ('CARGO_HOME','CARGO_TARGET_DIR','PYO3_PYTHON','TORCH_RS_CUDART','TORCH_RS_CUBLAS','TMPDIR','CUDA_CACHE_PATH','XDG_CACHE_HOME','TRITON_CACHE_DIR','TORCHINDUCTOR_CACHE_DIR','MPLCONFIGDIR','CUDA_VISIBLE_DEVICES','PYTHONPATH','PYTHONDONTWRITEBYTECODE')},
    orchestration_sha256=capture.sha256(__file__),environment_script_sha256=capture.sha256(OUT/'environment.sh')))
commands=[]

def run(label,command,mask='0'):
    clean()
    os.environ['CUDA_VISIBLE_DEVICES']=mask
    receipt=capture.capture(command,label)
    receipt.update(commit=COMMIT,measurement_kind='clean-code',git_status=git('status','--porcelain'),
                   cuda_visible_devices=mask,source_after=capture.source_provenance(),
                   extension_sha256=capture.sha256(ROOT/'python/torch_rs/torch_rs.abi3.so') if (ROOT/'python/torch_rs/torch_rs.abi3.so').exists() else None)
    capture.write(label+'.receipt.json',receipt)
    commands.append(receipt)
    capture.write('commands.json',commands)
    assert receipt['exit_status']==0,receipt
    assert hashes()==initial
    clean()
    print(label,'passed',flush=True)
