"""Audit and publish this review revision's development receipts unchanged."""
from pathlib import Path
import datetime, difflib, hashlib, json, shutil, subprocess, sys
ROOT=Path.cwd().resolve()
SRC=ROOT/'target/view-review'
HISTORY=ROOT/'docs/diagnostics/compile-cuda-view'
OUT=HISTORY/'review-revision'
def sha(p): return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def read(p): return json.loads(Path(p).read_text())
def put(p,obj): p.write_text(json.dumps(obj,indent=2,sort_keys=True)+'\n')
sys.path.insert(0,str(ROOT/'scripts'))
from evaluate_cuda_math import source_provenance
import torch_rs
source=source_provenance()
build=read(SRC/'release-final/build-record.json')
assert source['commit']=='4a0a77d4d73e70898e7b073dbc150a7d2a1a0562'
assert build['source_sha256']==source['source_sha256']
assert build['production_diff_sha256']==source['production_diff_sha256']
assert not build['clean_checkout'] and build['source_unchanged_during_build']
assert Path(sys.executable).resolve().is_relative_to(ROOT)
assert sha(torch_rs._C.__file__)==build['extension_sha256']
assert sha(build['wheel_path'])==build['wheel_sha256']
assert sha(build['interpreter_path'])==build['interpreter_sha256']
installed=Path(torch_rs.__file__).parent
python_files=list((ROOT/'python/torch_rs').rglob('*.py'))
assert all(sha(p)==sha(installed/p.relative_to(ROOT/'python/torch_rs')) for p in python_files)
history_counts={}
for folder in (HISTORY,HISTORY/'postcommit-12ac70fd'):
    inventory=read(folder/'inventory.json')
    assert all(sha(folder/p)==h for p,h in inventory.items())
    history_counts[str(folder.relative_to(ROOT))]=len(inventory)
base=read(HISTORY/'measured-inputs.json')
manifest_index={}
for p in (SRC/'evidence').glob('inputs-*.json'):
    data=read(p)
    manifest_index[sha(p)]={'replace':{k:v for k,v in data.items() if base.get(k)!=v},'remove':sorted(set(base)-set(data))}
receipts=list((SRC/'evidence').glob('*.receipt.json'))
final_names={'review-regressions-current-source','native-verification-final','compiler-current-source','portable-current-source','clippy-current-source','format-current-source','two-device','rust-graph','preflight','guide-examples'}
assert final_names <= {p.name.removesuffix('.receipt.json') for p in receipts}
for p in receipts:
    r=read(p)
    assert r['cwd']==str(ROOT)
    assert r['source_before']==r['source_after']
    assert r['git_status_before']==r['git_status_after']
    assert r['native_before']==r['native_after'] and r['inputs_unchanged']
    assert r['command_files_before']==r['command_files_after']
    for path,h in r['command_files_before'].items():
        assert Path(path).resolve().is_relative_to(ROOT)
        if sha(path)!=h:
            assert Path(path)==ROOT/'tests/test_compile_cuda_view.py'
            assert h in {sha(SRC/'first-tests.py'),sha(SRC/'focused-tests.py')}
    assert r['capture_script_sha256']==sha(SRC/'capture.py')
    assert r['measured_inputs_sha256'] in manifest_index
    assert r['log_sha256']==sha(SRC/'evidence'/p.name.replace('.receipt.json','.log'))
    for key,value in r['environment'].items():
        if isinstance(value,str) and value.startswith('/'): assert Path(value).resolve().is_relative_to(ROOT),(key,value)
    name=p.name.removesuffix('.receipt.json')
    assert r['exit_status']==(-2 if name=='compiler-final' else 0),(name,r['exit_status'])
    if name in final_names:
        assert r['source_before']==source
        assert set(r['native_before'].values())=={build['extension_sha256']}
        delta=manifest_index[r['measured_inputs_sha256']]
        inputs=base|delta['replace']
        for key in delta['remove']: inputs.pop(key)
        assert all(sha(ROOT/path)==h for path,h in inputs.items())
assert sha(SRC/'first-source.patch')==read(SRC/'release/build-record.json')['production_diff_sha256']
assert 'Ran 635 tests' in (SRC/'evidence/compiler-current-source.log').read_text()
assert 'OK (skipped=13)' in (SRC/'evidence/compiler-current-source.log').read_text()
assert 'skipping CUDA' not in (SRC/'evidence/rust-graph.log').read_text()
assert not OUT.exists()
OUT.mkdir()
for p in (SRC/'evidence').iterdir():
    if not p.name.startswith('inputs-'): shutil.copyfile(p,OUT/p.name)
for directory in ('release','release-final'):
    (OUT/directory).mkdir()
    for name in ('build-record.json','build.log','commands.json','install.log'):
        shutil.copyfile(SRC/directory/name,OUT/directory/name)
for name in ('capture.py','reproduce.py','compound_probe.py','compound_probe_verified.py','validate-final-gpu.sh','validate-final-portable.sh','publish.py'):
    shutil.copyfile(SRC/name,OUT/(name+'.txt'))
shutil.copyfile(ROOT/'target/view-review-setup.sh',OUT/'setup.sh.txt')
shutil.copyfile(SRC/'README.md',OUT/'README.md')
shutil.copyfile(SRC/'first-source.patch',OUT/'first-source.patch')
for label in ('first','focused'):
    after=(ROOT/'tests/test_compile_cuda_view.py').read_text()
    before=(SRC/(label+'-tests.py')).read_text()
    (OUT/(label+'-tests.patch')).write_text(''.join(difflib.unified_diff(after.splitlines(True),before.splitlines(True),fromfile='tests/test_compile_cuda_view.py',tofile='tests/test_compile_cuda_view.py')))
(OUT/'final-source.patch').write_bytes(subprocess.check_output(['git','diff','HEAD','--','python/torch_rs/_compile_bytecode.py','python/torch_rs/_compile_trace.py','tests/test_compile_cuda_view.py']))
put(OUT/'input-manifest-deltas.json',manifest_index)
put(OUT/'audit.json',{
    'measurement_kind':'review revision development; fresh clean-commit capture required after Burner commits',
    'audited_at':datetime.datetime.now(datetime.timezone.utc).isoformat(),
    'source':source,'cwd':str(ROOT),'git_status_at_publication':subprocess.check_output(['git','status','--porcelain'],text=True),
    'native_sha256':build['extension_sha256'],'wheel_sha256':build['wheel_sha256'],
    'interpreter':build['interpreter_path'],'all_installed_python_matches_source':True,'python_source_files_checked':len(python_files),
    'historical_inventories_verified_unchanged':history_counts,
    'manifest_base':'../measured-inputs.json','manifest_base_sha256':sha(HISTORY/'measured-inputs.json'),
    'final_commands':{p.name.removesuffix('.receipt.json'):{k:read(p)[k] for k in ('command','exit_status','started_at','ended_at','measured_inputs_sha256')} for p in receipts if p.name.removesuffix('.receipt.json') in final_names},
})
print('Published audited development evidence:',OUT)
