import sys
sys.path.insert(0,'.')
from tests.test_compile_cuda_view import *
for expression, expected in [('([1,2,3], foo=x)',TypeError), ('(2**63, k//1)',TypeError), ('(-2, True)',RuntimeError), ('((-1,-1,1))',RuntimeError)]:
    f=make_program(f'def program(x):\n    k = 1\n    return x.view{expression}\n')
    print(expression)
    for module in (native,torch):
        try: f(module.ones((1,)).to('cuda:0'))
        except Exception as e:
            print(module.__name__,type(e).__name__,str(e).splitlines()[0])
            assert type(e) is expected
        else: raise AssertionError('eager must reject')
    for policy in POLICIES:
        try: compile_with_cache(f,*policy)[0](native.ones((1,)).to('cuda:0'))
        except Exception as e:
            print(policy,type(e).__name__,str(e).splitlines()[0])
            assert type(e) is trace.CompileTraceUnsupportedError
        else: raise AssertionError('baseline must expose the gap')
