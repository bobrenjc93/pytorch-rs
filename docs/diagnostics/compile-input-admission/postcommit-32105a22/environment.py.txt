import hashlib, json, pathlib, subprocess, sys
import torch_rs as native
from torch_rs import torch_rs as bridge
root=pathlib.Path.cwd()
def fn(x): return -x
compiled=native.compile(fn)
output=compiled(native.ones(2).to('cuda:0'))
kernel=next(iter(compiled._torch_rs_pointwise_cache.executors.values()))
assert output.cpu().tolist()==[-1.,-1.]
print(json.dumps(dict(commit=subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip(),status=subprocess.check_output(['git','status','--porcelain'],text=True),cwd=str(root),executable=sys.executable,base_prefix=sys.base_prefix,package=native.__file__,extension=bridge.__file__,extension_sha256=hashlib.sha256(pathlib.Path(bridge.__file__).read_bytes()).hexdigest(),nvrtc_version=kernel.nvrtc_version,nvcc=subprocess.check_output(['nvcc','--version'],text=True),rustc=subprocess.check_output(['rustc','--version'],text=True),loaded_cuda_libraries=sorted({line.split()[-1] for line in pathlib.Path('/proc/self/maps').read_text().splitlines() if any(name in line for name in ('libcudart','libcuda.so'))}),smoke_values=output.cpu().tolist()),indent=2))
