"""Audit existing measured reports; no workloads or scoring settings are changed."""
import gzip, hashlib, importlib.util, json, math, pathlib, subprocess, sys, zipfile
root=pathlib.Path.cwd()
work=root/'target/postcommit-32105a22'
sys.path.insert(0,str(root/'scripts'))
import evaluate_torch_compile_default as evaluator

def digest(path):
    with path.open('rb') as stream:
        return hashlib.file_digest(stream,'sha256').hexdigest()
def git(where,*args): return subprocess.check_output(['git','-C',str(where),*args],text=True).strip()
reports={}
manifest={}
for label,checkout,report_path,export,commit in (
 ('candidate',root,work/'candidate-report.json',work/'candidate-export','32105a2277965dbf15a84cc464aaf3e5be0006d2'),
 ('main',work/'main',work/'main/target/postcommit/main-report.json',work/'main/target/postcommit/export','60202557b4f110d07777f585e804ab5f55e1ff7b')):
    report=json.loads(report_path.read_text())
    assert report['valid'] and not report['diagnostic']
    assert report['provenance']['commit']==git(checkout,'rev-parse','HEAD')==commit
    assert report['provenance']['working_tree_changes']==''
    assert git(checkout,'status','--porcelain','--untracked-files=all')==''
    for relative,sha in report['provenance']['files'].items():
        assert digest(checkout/relative)==sha,relative
    assert report['setup']['build_identity']=={key:report['provenance'][key] for key in ('commit','source_sha256','working_tree_changes')}
    assert len(report['workers'])==6 and len(report['cells'])==112
    assert report['warmups']==5 and report['samples']==17
    assert report['scores']['coverage']==evaluator.aggregate(report['cells'],'coverage')
    assert report['scores']['cuda-perf']==evaluator.aggregate([c for c in report['cells'] if c['device']=='cuda'],'cuda-perf')
    wheel=pathlib.Path(report['wheel']['path'])
    assert wheel.is_relative_to(checkout) and digest(wheel)==report['wheel']['sha256']
    with zipfile.ZipFile(wheel) as archive:
        for pyfile in (checkout/'python/torch_rs').rglob('*.py'):
            assert archive.read(str(pyfile.relative_to(checkout/'python')))==pyfile.read_bytes()
    for worker in report['workers']:
        assert len(worker['case_status'])==28
        if worker['implementation']=='reference':
            assert all(c['status']=='passed' for c in worker['case_status'].values())
        env=worker['environment']
        assert pathlib.Path(env['executable']).resolve().is_relative_to(checkout)
        assert digest(pathlib.Path(env['executable']))==env['interpreter_sha256']
        assert env['threads']==1 and env['cuda_visible_devices']=='0'
        if worker['implementation']=='candidate':
            assert pathlib.Path(env['package']).is_relative_to(checkout)
            assert digest(pathlib.Path(env['native_extension']))==env['native_sha256']
        else:
            for origin in env['reference_provenance']['origins'].values():
                assert pathlib.Path(origin['path']).is_relative_to(checkout)
                assert digest(pathlib.Path(origin['path']))==origin['sha256']
        if worker['device']=='cuda':
            sync=env['synchronization']
            assert pathlib.Path(sync['path']).is_relative_to(checkout)
            assert digest(pathlib.Path(sync['path']))==sync['sha256']
            assert sync['version']==13000 and sync['logical_device']==0
        for kind in ('log','output'):
            p=pathlib.Path(worker['artifacts'][kind])
            assert p.is_relative_to(checkout) and digest(p)==worker['artifacts'][kind+'_sha256']
    directory=pathlib.Path(report['run_directory'])
    entries=[]
    for path in sorted(directory.iterdir()):
        if path.is_dir(): continue
        assert path.is_file() and not path.is_symlink()
        assert digest(path)==digest(export/path.name)
        entries.append({'path':str(path),'export_path':str(export/path.name),'sha256':digest(path),'bytes':path.stat().st_size})
    manifest[label]=entries
    reports[label]=report
c,m=reports['candidate'],reports['main']
for path in ('scripts/evaluate_torch_compile_default.py','scripts/evaluate_torch_compile_default.sh','scripts/torch_compile_default_corpus.py','Cargo.toml','Cargo.lock','pyproject.toml','uv.lock'):
    assert c['provenance']['files'][path]==m['provenance']['files'][path]
key=lambda cell:(cell['device'],cell['case'],cell['variant'])
main_cells={key(cell):cell for cell in m['cells']}
rows=[]
changes=[]
for cell in c['cells']:
    other=main_cells[key(cell)]
    if cell['passed']!=other['passed']: changes.append(key(cell))
    if cell['device']=='cuda' and cell['passed'] and other['passed']:
        # Descriptive comparison of the existing two-order median samples;
        # does not replace or modify the frozen evaluator aggregation.
        native_ratio=math.prod(a['candidate']['median_ms']/b['candidate']['median_ms'] for a,b in zip(other['rounds'],cell['rounds']))**0.5
        ref_ratio=math.prod(a['reference']['median_ms']/b['reference']['median_ms'] for a,b in zip(other['rounds'],cell['rounds']))**0.5
        rows.append({'case':cell['case'],'variant':cell['variant'],'main_over_candidate_native_median':native_ratio,'main_over_candidate_reference_median':ref_ratio,'main_reference_over_native':other['ratio'],'candidate_reference_over_native':cell['ratio']})
result={'measured_commits':{label:r['provenance']['commit'] for label,r in reports.items()},'checks':'clean source, source manifest, wheel/Python bytes, installed imports, interpreter/runtime hashes, 112 cells, 6 workers, 28 cases per worker, successful references, frozen aggregation, exact export copies; all passed','scores':{label:r['scores'] for label,r in reports.items()},'pass_fail_changes':changes,'common_cuda_cells':rows,'common_cuda_geomean_main_over_candidate_native_median':math.exp(sum(math.log(r['main_over_candidate_native_median']) for r in rows)/len(rows)),'common_cuda_geomean_main_over_candidate_reference_median':math.exp(sum(math.log(r['main_over_candidate_reference_median']) for r in rows)/len(rows))}
(work/'audit.json').write_text(json.dumps(result,indent=2)+'\n')
(work/'raw-retention-manifest.json').write_text(json.dumps(manifest,indent=2)+'\n')
print(json.dumps({k:v for k,v in result.items() if k not in ('scores','common_cuda_cells')},indent=2))
